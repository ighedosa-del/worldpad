    88→

    90→    console.warn('[GlobalAI] SIMULATION TRADE — NOT sending to Deriv. isAuthorized=', authorized);
    91→    addAutoTraderLog(`⚠️ SIM TRADE: ${params.contractType} ${params.symbol} $${params.stake} — NOT on Deriv`);
    92→    if (pendingSimTradesGlobal.has(params.symbol)) {
    93→      console.warn('[GlobalAI] SIM TRADE DROPPED: pending already exists for', params.symbol);
    94→      return null;
    95→    }
    96→
    97→    return new Promise<TradeResult | null>((resolve) => {
    98→      const timeout = setTimeout(() => {
    99→        pendingSimTradesGlobal.delete(params.symbol);
   100→        resolve(null);
   101→      }, 15000);
   102→
   103→      registerGlobalTickListener();
   104→      pendingSimTradesGlobal.set(params.symbol, {
   105→        ...params,
   106→        resolve: (r) => { clearTimeout(timeout); resolve(r); },
   107→      });
   108→    });
   109→  }
   110→
   111→  // LIVE mode — sending real trade to Deriv
   112→  try {
   113→    const wsBefore = getTradeWSStatus();
   114→    useWorldpadStore.getState().addAutoTraderLog(`[AI] 🔄 LIVE TRADE — connecting to Deriv... (ws ready: ${wsBefore.wsReady})`);
   115→    console.log('[GlobalAI] LIVE TRADE — sending to Deriv WS. Status:', wsBefore);
   116→
   117→    const proposal = await getProposalWS({
   118→      contractType: params.contractType,
   119→      symbol: params.symbol,
   120→      stake: params.stake,
   121→      barrier: params.barrier,
   122→      duration: 1,
   123→      durationUnit: 't',
   124→    });
   125→    useWorldpadStore.getState().addAutoTraderLog(`[AI] ✅ Proposal OK — $${proposal.payout.toFixed(2)} payout, buying...`);
   126→

   128→    const won = buyResult.profit > 0;
   129→    useWorldpadStore.getState().addAutoTraderLog(`[AI] ✅ BUY FILLED — contract ${buyResult.contract_id} | ${won ? '+$' : '-$'}${Math.abs(buyResult.profit).toFixed(2)} ${won ? 'WIN' : 'LOSS'} [LIVE ON DERIV]`);
   130→    return {
   131→      id: buyResult.contract_id,
   132→      type: params.contractType,
   133→      symbol: params.symbol,
   134→      stake: params.stake,
   135→      payout: buyResult.payout,
   136→      profit: buyResult.profit,
   137→      digit: params.barrier ?? -1,
   138→      won,
   139→      timestamp: Date.now(),
   140→      simulated: false,
   141→    };
   142→  } catch (err) {
   143→    const errMsg = (err as Error).message || String(err);
   144→    const wsAfter = getTradeWSStatus();
   145→    console.error('[GlobalAI] Live trade error:', errMsg, 'ws after:', wsAfter);
   146→    useWorldpadStore.getState().addAutoTraderLog(`[AI] ❌ TRADE FAILED: ${errMsg}`);
   147→    if (errMsg.includes('timed out')) {
   148→      useWorldpadStore.getState().addAutoTraderLog(`[AI] ❌ Trade timed out — check PAT token has TRADE scope in Deriv Settings`);
   149→    }
   150→    if (errMsg.includes('Not authenticated')) {
   151→      useWorldpadStore.getState().addAutoTraderLog(`[AI] ❌ Not connected — click Connect Account`);
   152→    }
   153→    return null;
   154→  }
