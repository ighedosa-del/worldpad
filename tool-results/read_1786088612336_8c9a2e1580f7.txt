    27→let globalTickListenerRegistered = false;
    28→
    29→function registerGlobalTickListener() {
    30→  if (globalTickListenerRegistered) return;
    31→  globalTickListenerRegistered = true;
    32→
    33→  addTickListener((symbol, data) => {
    34→    const pending = pendingSimTradesGlobal.get(symbol);
    35→    if (!pending || !data.lastTick) return;
    36→
    37→    const nextDigit = data.lastTick.digit;
    38→    pendingSimTradesGlobal.delete(symbol);
    39→
    40→    const { contractType, barrier, stake, symbol: tradeSymbol } = pending;
    41→    let won = false;
    42→    switch (contractType) {
    43→      case 'DIGITMATCH': won = nextDigit === barrier; break;
    44→      case 'DIGITDIFF': won = nextDigit !== barrier; break;
    45→      case 'DIGITOVER': won = nextDigit > (barrier ?? 4); break;
    46→      case 'DIGITUNDER': won = nextDigit < (barrier ?? 5); break;
    47→      case 'DIGITEVEN': won = nextDigit % 2 === 0; break;
    48→      case 'DIGITODD': won = nextDigit % 2 === 1; break;
    49→      default: won = Math.random() > 0.5;
    50→    }
    51→
    52→    // Payout: Deriv pays net profit multiplier on win, 0 on loss
    53→    const isMatch = contractType === 'DIGITMATCH';
    54→    const profitMultiplier = isMatch ? 8.5 : 0.85;
    55→    const profit = won ? stake * profitMultiplier : -stake;
    56→    const payout = won ? stake + profit : 0;
    57→
    58→    const result: TradeResult = {
    59→      id: `SIM-${Date.now()}`,
    60→      type: contractType,
    61→      symbol: tradeSymbol,
    62→      stake,
    63→      payout,
    64→      profit,
    65→      digit: barrier ?? -1,
    66→      won,
    67→      timestamp: Date.now(),
    68→      simulated: true,
    69→    };
    70→
    71→    pending.resolve(result);
    72→  });
    73→}

    75→async function placeTradeDirect(params: {
    76→  contractType: string;
    77→  barrier?: number;
    78→  stake: number;
    79→  symbol: string;
    80→}): Promise<TradeResult | null> {
    81→  const authorized = useWorldpadStore.getState().isAuthorized;
    82→  const logFn = useWorldpadStore.getState().addAutoTraderLog;
    83→
    84→  // === LIVE MODE ===
    85→  if (authorized) {
    86→    try {
    87→      const wsBefore = getTradeWSStatus();
    88→      logFn(`[LIVE] ${params.contractType} ${params.symbol} $${params.stake} d${params.barrier ?? '-'} | ws=${wsBefore.wsReady} auth=${wsBefore.authorized}`);
    89→
    90→      const proposal = await getProposalWS({
    91→        contractType: params.contractType,
    92→        symbol: params.symbol,
    93→        stake: params.stake,
    94→        barrier: params.barrier,
    95→        duration: 1,
    96→        durationUnit: 't',
    97→      });
    98→
    99→      const buyResult = await buyContractWS(proposal.id, proposal.ask_price);
   100→      const won = buyResult.profit > 0;
   101→      logFn(`[LIVE] ${won ? 'WIN' : 'LOSS'} $${Math.abs(buyResult.profit).toFixed(2)} contract=${buyResult.contract_id}`);
   102→      return {
   103→        id: buyResult.contract_id,
   104→        type: params.contractType,
   105→        symbol: params.symbol,
   106→        stake: params.stake,
   107→        payout: buyResult.payout,
   108→        profit: buyResult.profit,
   109→        digit: params.barrier ?? -1,
   110→        won,
   111→        timestamp: Date.now(),
   112→        simulated: false,
   113→      };
   114→    } catch (err) {
   115→      const errMsg = (err as Error).message || String(err);
   116→      logFn(`[LIVE FAILED] ${errMsg} → instant SIM fallback`);
   117→      // Fall through to SIM below
   118→    }
   119→  }
   120→
   121→  // === SIM MODE (or LIVE fallback) ===
   122→  // v10: Instant resolution — no waiting for ticks
   123→  const md = getMarketData(params.symbol);
   124→  let nextDigit: number;
   125→  if (md?.lastTick) {
   126→    nextDigit = md.lastTick.digit;
   127→  } else {
   128→    nextDigit = Math.floor(Math.random() * 10);
   129→  }
   130→
   131→  const { contractType, barrier, stake } = params;
   132→  let won = false;
   133→  switch (contractType) {
   134→    case 'DIGITMATCH': won = nextDigit === barrier; break;
   135→    case 'DIGITDIFF': won = nextDigit !== barrier; break;
   136→    case 'DIGITOVER': won = nextDigit > (barrier ?? 4); break;
   137→    case 'DIGITUNDER': won = nextDigit < (barrier ?? 5); break;
   138→    case 'DIGITEVEN': won = nextDigit % 2 === 0; break;
   139→    case 'DIGITODD': won = nextDigit % 2 === 1; break;
   140→    default: won = Math.random() > 0.5;
   141→  }
   142→
   143→  const isMatch = contractType === 'DIGITMATCH';
   144→  const profitMultiplier = isMatch ? 8.5 : 0.85;
   145→  const profit = won ? stake * profitMultiplier : -stake;
   146→  const payout = won ? stake + profit : 0;
   147→
   148→  logFn(`[SIM] ${contractType} ${params.symbol} d${barrier ?? '-'} $${stake} → ${won ? 'WIN +$' + profit.toFixed(2) : 'LOSS -$' + stake.toFixed(2)} (digit=${nextDigit})`);
   149→
   150→  return {
   151→    id: `SIM-${Date.now()}`,
   152→    type: contractType,
   153→    symbol: params.symbol,
   154→    stake,
   155→    payout,
   156→    profit,
   157→    digit: barrier ?? -1,
   158→    won,
   159→    timestamp: Date.now(),
   160→    simulated: true,
   161→  };
   162→}

   332→          market.symbol, signal.contractType, signal.barrier,

   340→          id: `ai-${Date.now()}-${market.symbol}`,
   341→          type: signal.contractType,
   342→          symbol: market.symbol,
   343→          stake: finalStake,
   344→          payout: result.payout || finalStake * 0.85,
   345→          profit: result.profit,
   346→          digit: signal.barrier ?? -1,

   348→          timestamp: Date.now(),

   490→    const simMode = !authorized;
   491→
   492→    // v5: Pre-flight safety check for live trading

   500→      if (runningRef.current) {
   501→        // v8: 2s cycle — faster trading
   502→        cycleTimerRef.current = setTimeout(runLoop, 2000);
   503→      }
   504→    };
   505→    runLoop();
   506→  }, [isAuthorized, botConfig, addAutoTraderLog, runCycle, setGlobalAIRunning, setGlobalAICycleCount, setGlobalAITotalTrades, setGlobalAITotalProfit, setGlobalAILearningStats]);
   507→
   508→  const stopBot = useCallback(() => {
   509→    runningRef.current = false;
   510→    setGlobalAIRunning(false);
   511→    setGlobalAIStatus('idle');
   512→    if (cycleTimerRef.current) { clearTimeout(cycleTimerRef.current); cycleTimerRef.current = null; }

   514→    tradeLocksRef.current.clear();

   519→    addAutoTraderLog(`[AI] Session W/L: ${sessionWinsRef.current}/${sessionLossesRef.current}`);
   520→    addAutoTraderLog(`[AI] ═══════════════════════════════════════`);
   521→    aiEngine.saveLearningData();
   522→  }, [addAutoTraderLog, setGlobalAIRunning, setGlobalAIStatus]);
   523→
   524→  // Expose start/stop for AI Scanner buttons
   525→  useEffect(() => {
   526→    (window as any).__globalAI = { startBot, stopBot };
   527→    return () => { delete (window as any).__globalAI; };
   528→  }, [startBot, stopBot]);
   529→
   530→  // === Init: start scanning + auto-start bot ===
   531→  useEffect(() => {
   532→    mountedRef.current = true;
   533→    aiEngine.loadLearningData();
   534→    setGlobalAILearningStats(aiEngine.getLearningStats());
   535→    startMultiMarketScan();
   536→
   537→    // v11: Restore Deriv credentials and start bot regardless of mode
   538→    const store = useWorldpadStore.getState();
   539→    console.log('[GlobalAI] INIT — isAuthorized:', store.isAuthorized, 'mode:', store.accountMode, 'hasDemoToken:', !!store.demoToken, 'hasRealToken:', !!store.realToken, 'accountId:', store.selectedAccountId);
   540→
   541→    const initAndStart = async () => {
   542→      // v12: Hardcoded token from .env.local — always connect LIVE
   543→      const envToken = process.env.NEXT_PUBLIC_DERIV_TOKEN;
   544→      const envAppId = process.env.NEXT_PUBLIC_DERIV_APP_ID || '1089';
   545→      let connected = false;
   546→
   547→      if (envToken) {
   548→        console.log('[GlobalAI] Using hardcoded token from .env.local');
   549→        const result = await restoreCredentials(envToken, envAppId);
   550→        if (result) {
   551→          console.log('[GlobalAI] ✅ LIVE connected:', result.loginid, 'balance:', result.balance, 'type:', result.accountType);
   552→          // Force store into authorized LIVE state

   557→          s.setSelectedAccountId(result.loginid);
   558→          if (result.accountType === 'demo') s.setDemoToken(envToken); else s.setRealToken(envToken);
   559→          s.setDerivAppId(envAppId);
   560→          s.setIsAuthorized(true);
   561→          connected = true;
   562→        } else {
   563→          console.error('[GlobalAI] ❌ Hardcoded token FAILED to connect');
   564→        }
   565→      } else if (store.isAuthorized && (store.demoToken || store.realToken)) {
   566→        // Fallback: use token from store (user logged in via modal)
   567→        const token = store.accountMode === 'demo' ? store.demoToken : store.realToken;
   568→        if (token) {
   569→          const result = await restoreCredentials(token, store.derivAppId || '1089');
   570→          if (result) {
   571→            useWorldpadStore.getState().setBalance(result.balance);
   572→            useWorldpadStore.getState().setAccountInfo({ fullname: result.fullname, loginid: result.loginid, balance: result.balance, currency: result.currency });
   573→            connected = true;
   574→          }
   575→        }
   576→      }
   577→
   578→      // ALWAYS auto-start the bot
   579→      if (mountedRef.current && !runningRef.current) {
   580→        const isLive = useWorldpadStore.getState().isAuthorized;
   581→        console.log('[GlobalAI] Starting bot... mode=', isLive ? 'LIVE' : 'SIM');
   582→        addAutoTraderLog(`[AI] ═══════════════════════════════════════`);
   583→        addAutoTraderLog(`[AI] === BOT STARTED === mode=${isLive ? 'LIVE' : 'SIM'}${connected ? ' ✅ CONNECTED' : ''}`);
   584→        addAutoTraderLog(`[AI] ═══════════════════════════════════════`);
   585→        startBot();
   586→      }
   587→    };
   588→
   589→    // Wait 5s for scanner to collect some data, then start
   590→    const autoStartTimer = setTimeout(initAndStart, 5000);
   591→
   592→    return () => {
   593→      mountedRef.current = false;
   594→      runningRef.current = false;
   595→      if (cycleTimerRef.current) clearTimeout(cycleTimerRef.current);
   596→      if (rankingTimerRef.current) clearTimeout(rankingTimerRef.current);
   597→      clearTimeout(autoStartTimer);
   598→      pendingSimTradesGlobal.clear();
   599→      stopMultiMarketScan();
   600→    };
   601→  }, []);
   602→
   603→  // Health monitoring
   604→  useEffect(() => {
   605→    const interval = setInterval(() => {
   606→      if (!mountedRef.current) return;
   607→      const connected = isScannerConnected();
   608→      const health = getScannerHealth();
   609→      if (mountedRef.current) {
   610→        setGlobalAIHealth({
   611→          isConnected: connected,
   612→          totalTicksReceived: health.totalTicksReceived,
   613→          lastTickTime: health.lastTickTime,
   614→          connectTime: health.connectTime,
   615→          ticksPerMarket: health.ticksPerMarket,
   616→          wsError: health.wsError,
   617→          callbackCount: health.callbackCount,
   618→        });
   619→      }
   620→    }, 1000);
   621→    return () => clearInterval(interval);
   622→  }, [setGlobalAIHealth]);
   623→
   624→  // v5: Auto-restart bot when user connects mid-session
   625→  // CRITICAL FIX: If bot is already running in SIM mode, we MUST stop + restart it
   626→  // so it picks up the new isAuthorized=true closure for LIVE trades.
   627→  useEffect(() => {
   628→    if (!isAuthorized) return;
   629→
   630→    const store = useWorldpadStore.getState();
   631→    const token = store.accountMode === 'demo' ? store.demoToken : store.realToken;
   632→    if (token) {
   633→      restoreCredentials(token, '1089');
   634→    }
   635→
   636→    // Start the bot (with credentials restored above)
   637→    if (runningRef.current) {
   638→      runningRef.current = false;
   639→      if (cycleTimerRef.current) { clearTimeout(cycleTimerRef.current); cycleTimerRef.current = null; }
   640→      activeTradesRef.current.clear();
   641→      tradeLocksRef.current.clear();
   642→      pendingSimTradesGlobal.clear();
   643→      setGlobalAIRunning(false);
   644→    }
   645→    addAutoTraderLog(`[AI] 🔗 Account connected — starting bot (${store.accountMode})...`);
   646→    setTimeout(() => { startBot(); }, 1000);
   647→  }, [isAuthorized, startBot, addAutoTraderLog, setGlobalAIRunning]);
   648→
   649→  // Listen for trades from ANY tab and feed AI learning
   650→  useEffect(() => {
   651→    const unsub = useWorldpadStore.subscribe(
   652→      (state) => state.tradeHistory,
   653→      (history) => {
   654→        if (history.length === 0) return;
   655→        const last = history[history.length - 1];
   656→        if (last.id.startsWith('ai-')) return;
   657→        aiEngine.recordTradeResult(
   658→          last.symbol, last.type, last.digit >= 0 ? last.digit : undefined,
   659→          last.profit, 50
   660→        );
   661→      }
   662→    );
   663→    return unsub;
   664→  }, []);
   665→
   666→  return null;
