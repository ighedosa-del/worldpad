   168→function playTone(freq: number, duration: number, type: OscillatorType = 'sine', vol = 0.15) {
   169→  try {
   170→    const ctx = getAudioCtx();
   171→    const osc = ctx.createOscillator();
   172→    const gain = ctx.createGain();
   173→    osc.type = type;
   174→    osc.frequency.value = freq;
   175→    gain.gain.value = vol;
   176→    gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + duration);
   177→    osc.connect(gain);
   178→    gain.connect(ctx.destination);
   179→    osc.start();
   180→    osc.stop(ctx.currentTime + duration);
   181→  } catch {}
   182→}

   189→// DIFFER IS BANNED — 0.09x payout means you lose $0.91 per loss vs gain $0.03 per win
   190→function signalToContract(subType: string, prediction: number | null): { contractType: string; barrier: string } | null {
   191→  const t = subType.toUpperCase();
   192→  switch (t) {
   193→    case 'MATCH': case 'DIGIT MATCH': return prediction !== null ? { contractType: 'DIGITMATCH', barrier: String(prediction) } : null;
   194→    // DIFFER BANNED — returns null, trade will be skipped
   195→    case 'DIFFER': case 'DIGIT DIFF': return null; // BANNED: 0.09x payout = guaranteed loss
   196→    case 'OVER': case 'DIGIT OVER': return prediction !== null ? { contractType: 'DIGITOVER', barrier: String(prediction) } : null;
   197→    case 'UNDER': case 'DIGIT UNDER': return prediction !== null ? { contractType: 'DIGITUNDER', barrier: String(prediction) } : null;
   198→    case 'EVEN': return { contractType: 'DIGITEVEN', barrier: '' };
   199→    case 'ODD': return { contractType: 'DIGITODD', barrier: '' };
   200→    case 'RISE': return { contractType: 'CALL', barrier: '' };
   201→    case 'FALL': return { contractType: 'PUT', barrier: '' };
   202→    default: return null;
   203→  }
   204→}

   206→// ── Stake Calculator (Simple + Anti-Martingale) ──
   207→// Keep it simple: base stake most of the time
   208→// Reduce after 3+ consecutive losses (survival)
   209→// Increase after 3+ consecutive wins (ride the wave)
   210→function calcStake(baseStake: number, consecutiveWins: number, consecutiveLosses: number, accountBalance: number, todayProfit: number, profitLockMode: boolean): number {
   211→  let stake = baseStake;
   212→  // Reduce after losses (but not too aggressive — was killing profits)
   213→  if (consecutiveLosses >= 4) stake = baseStake * 0.5;
   214→  else if (consecutiveLosses >= 3) stake = baseStake * 0.75;
   215→  // Increase after wins
   216→  else if (consecutiveWins >= 4) stake = baseStake * 1.4;
   217→  else if (consecutiveWins >= 3) stake = baseStake * 1.25;
   218→  else if (consecutiveWins >= 2) stake = baseStake * 1.15;
   219→  else if (consecutiveWins >= 1) stake = baseStake * 1.05;
   220→  // Profit lock
   221→  if (profitLockMode && todayProfit >= 2) {
   222→    stake = Math.min(stake, baseStake * 0.5);
   223→  }
   224→  // Safety: never risk more than 5% of balance, min $0.10
   225→  const maxStake = Math.min(accountBalance * 0.05, 5);
   226→  return Math.round(Math.max(stake, 0.10) * 100) / 100;
   227→}

   528→      const last5 = recentResults.slice(-5);
   529→      const recentWins = last5.filter(r => r === 1).length;
   530→      const recentLosses = last5.filter(r => r === 0).length;
   531→      const recentWinRate = last5.length > 0 ? Math.round((recentWins / last5.length) * 100) : 50;
   532→
   533→      // ── HARD BLOCK: skip combos that are actively losing ──
   534→      // Block if: last 3+ results are ALL losses (hot streak of losses)
   535→      if (last5.length >= 3) {
   536→        const last3 = last5.slice(-3);
   537→        if (last3.every(r => r === 0)) return; // 3 consecutive losses → skip entirely
   538→      }
   539→
   540→      // ── TIMEOUT after loss: wait longer if combo is losing ──
   541→      let cooldownTime = 2000;
   542→      if (comboTotal >= 2) {
   543→        if (comboWinRate < 35) cooldownTime = 30000;      // Terrible: 30s cooldown
   544→        else if (comboWinRate < 45) cooldownTime = 15000;  // Bad: 15s cooldown
   545→        else if (recentWinRate < 30 && last5.length >= 3) cooldownTime = 20000; // Recent slump
   546→      }
   547→
   548→      // ── DYNAMIC CONFIDENCE THRESHOLD based on learning ──
   549→      // MATCH: starts at 45%, adjusts with learning
   550→      // Others: start at 65%, adjust with learning
   551→      let dynamicThreshold = tab === 'matches_differs' ? 45 : 65;
   552→
   553→      if (comboTotal >= 2) {
   554→        if (comboWinRate < 30) dynamicThreshold += 20;        // Terrible: need 90%+ to retry
   555→        else if (comboWinRate < 40) dynamicThreshold += 12;   // Bad: need much higher
   556→        else if (comboWinRate < 50) dynamicThreshold += 5;    // Below avg: slightly higher
   557→        else if (comboWinRate >= 65) dynamicThreshold -= 5;   // Good: lower bar
   558→        else if (comboWinRate >= 75) dynamicThreshold -= 10;  // Great: much lower bar
   559→      }
   560→
   561→      // Recent streak adjustment
   562→      if (last5.length >= 3) {
   563→        if (recentWinRate <= 20) dynamicThreshold += 15;      // Recent disaster
   564→        else if (recentWinRate <= 40) dynamicThreshold += 5;   // Recent slump
   565→      }
   566→
   567→      // Apply the dynamic threshold
   568→      if (signal.confidence < dynamicThreshold) return;
   569→
   570→      // ── Cooldown check (after dynamic threshold) ──
   571→      const lastTradeTime = pendingTradesRef.current.get(comboKey) ?? 0;
   572→      if (Date.now() - lastTradeTime < cooldownTime) return;
   573→
   574→      const mapped = signalToContract(signal.subType, signal.prediction);
   575→      if (!mapped) return;
   576→
   577→      // ═══ EV FILTER (simplified — no Bayesian complexity) ═══
   578→      const payout = PAYOUT_MULTIPLIERS[mapped.contractType] ?? 0.85;
   579→      // Use learned win rate if available (5+ trades), otherwise signal confidence
   580→      const effectiveWinProb = comboTotal >= 5
   581→        ? comboWins / comboTotal
   582→        : signal.confidence / 100;
   583→      const ev = (effectiveWinProb * payout) - ((1 - effectiveWinProb) * 1);
   584→
   585→      // Block negative EV trades
   586→      if (ev < MIN_EV_THRESHOLD) return;
   587→
   588→      // MATCH gets a lower EV threshold since payout is huge
   589→      // Even at 12% WR: EV = 0.12 * 9 - 0.88 = +0.20 (very profitable)
   590→      // So for MATCH, we use a much lower EV bar
   591→      if (mapped.contractType === 'DIGITMATCH' && ev >= -0.5) {
   592→        // Allow slightly negative EV for MATCH — the 9x payout makes variance work in our favor
   593→        // A MATCH that hits at 11% WR breaks even. At 15% WR it's hugely profitable.
   594→        // The learning engine will shut down combos that consistently lose.
   595→      }
   596→
   597→      pendingTradesRef.current.set(comboKey, Date.now());
   598→      tradesThisCycle++;
   599→      const tradeId = `${symbol}_${tab}_${Date.now()}`;
   600→
   601→      // ── Classify learning strength (fast: starts after 2 trades) ──
   602→      let learnedStrength: string;
   603→      if (comboTotal === 0) learnedStrength = 'NEW';
   604→      else if (comboTotal < 2) learnedStrength = 'LEARNING';
   605→      else if (comboWinRate >= 70) learnedStrength = 'VERY STRONG';
   606→      else if (comboWinRate >= 60) learnedStrength = 'STRONG';
   607→      else if (comboWinRate >= 50) learnedStrength = 'MODERATE';
   608→      else learnedStrength = 'WEAK';
   609→
   610→      // ── Learning boost (aggressive: starts after 2 trades) ──
   611→      let learningBoost = 0;
   612→      if (comboTotal >= 2) {
   613→        if (comboWinRate >= 75) learningBoost = 8;
   614→        else if (comboWinRate >= 65) learningBoost = 5;
   615→        else if (comboWinRate >= 55) learningBoost = 2;
   616→        else if (comboWinRate < 35) learningBoost = -10;
   617→        else if (comboWinRate < 45) learningBoost = -5;
   618→      }
   619→      const adjustedConfidence = Math.min(100, Math.max(0, signal.confidence + learningBoost));
   620→
   621→      const record: TradeRecord = {
   622→        id: tradeId, contractId: null, symbol,
   623→        symbolName: symbolNameMap[symbol] || symbol,
   624→        tradeType: mapped.contractType, prediction: mapped.barrier || signal.subType,
   625→        confidence: signal.confidence, qualityScore: signal.qualityScore,
   626→        stake, status: 'PENDING', profit: null, payout: null, buyPrice: null,
   627→        longcode: '', timestamp: Date.now(), tab,
   628→        learnedWinRate: comboWinRate,
   629→        learnedWins: comboWins,
   630→        learnedLosses: comboLosses,
   631→        learnedStrength,
   632→        adjustedConfidence,
   633→        expectedValue: ev,
   634→        payoutMultiplier: payout,
   635→      };
   636→
   637→      setTrades(prev => [record, ...prev].slice(0, 200));
   638→      const learnTag = comboTotal > 0
   639→        ? ` [LRN: ${comboWinRate}% ${comboWins}W/${comboLosses}L ${learnedStrength}]`
   640→        : ' [LRN: NEW]';
   641→      const thresholdTag = comboTotal >= 2 ? ` (bar:${dynamicThreshold}%)` : '';
   642→      const evTag = ` [EV:${(ev * 100).toFixed(0)}% pay:${payout}x]`;
   643→      setStatusMsg(`${signal.subType} ${signal.prediction ?? ''} on ${symbolNameMap[symbol]} [${TAB_LABELS[tab] || tab}] $${stake.toFixed(2)} [${signal.confidence}%${learningBoost !== 0 ? (learningBoost > 0 ? '+' : '') + learningBoost + '%' : ''}]${learnTag}${thresholdTag}${evTag}`);
   644→
   645→      clientRef.current!.placeTrade(symbol, mapped.contractType, mapped.barrier, stake, signal.ticksRecommendation, 't')
   646→        .then(result => {
   647→          if (!result.success) {
   648→            setTrades(prev => prev.map(t => t.id === tradeId ? { ...t, status: 'ERROR', error: result.error } : t));
   649→            pendingTradesRef.current.delete(comboKey);
   650→          }
   651→        });
   652→    });
   653→  }, [account]);
   654→
   655→  const processSignals = processAllSignals;
   656→
   657→  const emergencyStop = useCallback(() => {
   658→    setSettings(prev => ({ ...prev, autoTrading: false }));
   659→    setStatusMsg('EMERGENCY STOP');
   660→  }, []);
   661→
   662→  return {
   663→    settings, setSettings, account, connecting, connectError,
   664→    connect, disconnect, trades, stats, statusMsg,
   665→    processSignals, processAllSignals, emergencyStop,
   666→    comboStats, plHistory, consecutiveWins,
   667→    isConnected: !!account,
   668→  };
   669→}
