   181→function HealthBanner({ health, isRunning }: { health: ScannerHealth; isRunning: boolean }) {
   182→  const secondsSinceConnect = health.connectTime > 0 ? Math.floor((Date.now() - health.connectTime) / 1000) : 0;
   183→  const totalTicks = Object.values(health.ticksPerMarket).reduce((a, b) => a + b, 0);
   184→  const noTicks = secondsSinceConnect > 5 && totalTicks === 0;
   185→  const hasError = !!health.wsError;
   186→  if (!isRunning && !hasError && totalTicks === 0) return null;
   187→  if (!noTicks && !hasError && totalTicks > 0) return null;
   188→  return (
   189→    <div className="flex items-center gap-2 px-3 py-2 rounded-lg" style={{
   190→      background: hasError ? 'rgba(239,68,68,0.08)' : 'rgba(234,179,8,0.08)',
   191→      border: `1px solid ${hasError ? 'rgba(239,68,68,0.2)' : 'rgba(234,179,8,0.15)'}`,
   192→    }}>
   193→      <AlertTriangle className="w-4 h-4 shrink-0" style={{ color: hasError ? '#ef4444' : '#eab308' }} />
   194→      <div className="flex-1 min-w-0">
   195→        <div className="text-xs font-medium" style={{ color: hasError ? '#fca5a5' : '#fde68a' }}>
   196→          {hasError ? `Connection issue: ${health.wsError}` : noTicks ? `Connected ${secondsSinceConnect}s ago, waiting for data...` : 'Check connection'}
   197→        </div>
   198→        <div className="text-[10px] text-gray-500 mt-0.5">WS: {health.isConnected ? 'OPEN' : 'CLOSED'} | Ticks: {totalTicks} | Callbacks: {health.callbackCount}</div>
   199→      </div>
   200→    </div>
   201→  );
   202→}
