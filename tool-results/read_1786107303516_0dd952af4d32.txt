    27→let globalTickListenerRegistered = false;
    28→
    29→function registerGlobalTickListener() {
    30→  if (globalTickListenerRegistered) return;
    31→  globalTickListenerRegistered = true;
    32→
    33→  addTickListener((symbol, data) => {
    34→    const pending = pendingSimTradesGlobal.get(symbol);
    35→    if (!pending || !data.lastTick) return;
    36→
    37→    const nextDigit = data.lastTick.digit;
    38→    pendingSimTradesGlobal.delete(symbol);
    39→
    40→    const { contractType, barrier, stake, symbol: tradeSymbol } = pending;
    41→    let won = false;
    42→    switch (contractType) {
    43→      case 'DIGITMATCH': won = nextDigit === barrier; break;
    44→      case 'DIGITDIFF': won = nextDigit !== barrier; break;
    45→      case 'DIGITOVER': won = nextDigit > (barrier ?? 4); break;
    46→      case 'DIGITUNDER': won = nextDigit < (barrier ?? 5); break;
    47→      case 'DIGITEVEN': won = nextDigit % 2 === 0; break;
    48→      case 'DIGITODD': won = nextDigit % 2 === 1; break;
    49→      default: won = Math.random() > 0.5;
    50→    }
    51→
    52→    // Payout: Deriv pays net profit multiplier on win, 0 on loss
    53→    const isMatch = contractType === 'DIGITMATCH';
    54→    const profitMultiplier = isMatch ? 8.5 : 0.85;
    55→    const profit = won ? stake * profitMultiplier : -stake;
    56→    const payout = won ? stake + profit : 0;
    57→
    58→    const result: TradeResult = {
    59→      id: `SIM-${Date.now()}`,
    60→      type: contractType,
    61→      symbol: tradeSymbol,
    62→      stake,
    63→      payout,
    64→      profit,
    65→      digit: barrier ?? -1,
    66→      won,
    67→      timestamp: Date.now(),
    68→      simulated: true,
    69→    };
    70→
    71→    pending.resolve(result);
    72→  });
    73→}

    75→async function placeTradeDirect(params: {
    76→  contractType: string;
    77→  barrier?: number;
    78→  stake: number;
    79→  symbol: string;
    80→}): Promise<TradeResult | null> {
    81→  const authorized = useWorldpadStore.getState().isAuthorized;
    82→  const logFn = useWorldpadStore.getState().addAutoTraderLog;
    83→
    84→  // === LIVE MODE ===
    85→  if (authorized) {
    86→    try {
    87→      const wsBefore = getTradeWSStatus();
    88→      logFn(`[LIVE] ${params.contractType} ${params.symbol} $${params.stake} d${params.barrier ?? '-'} | ws=${wsBefore.wsReady} auth=${wsBefore.authorized}`);
    89→
    90→      const proposal = await getProposalWS({
    91→        contractType: params.contractType,
    92→        symbol: params.symbol,
    93→        stake: params.stake,
    94→        barrier: params.barrier,
    95→        duration: 1,
    96→        durationUnit: 't',
    97→      });
    98→
    99→      const buyResult = await buyContractWS(proposal.id, proposal.ask_price);
   100→      const won = buyResult.profit > 0;
   101→      logFn(`[LIVE] ${won ? 'WIN' : 'LOSS'} $${Math.abs(buyResult.profit).toFixed(2)} contract=${buyResult.contract_id}`);
   102→      return {
   103→        id: buyResult.contract_id,
   104→        type: params.contractType,
   105→        symbol: params.symbol,
   106→        stake: params.stake,
   107→        payout: buyResult.payout,
   108→        profit: buyResult.profit,
   109→        digit: params.barrier ?? -1,
   110→        won,
   111→        timestamp: Date.now(),
   112→        simulated: false,
   113→      };
   114→    } catch (err) {
   115→      const errMsg = (err as Error).message || String(err);
   116→      logFn(`[LIVE FAILED] ${errMsg} → SIM fallback`);
   117→      // Fall through to SIM below
   118→    }
   119→  }
   120→
   121→  // === SIM MODE (or LIVE fallback) ===
   122→  try {
   123→    const md = getMarketData(params.symbol);
   124→    let nextDigit: number;
   125→    if (md?.lastTick) {
   126→      nextDigit = md.lastTick.digit;
   127→    } else {
   128→      nextDigit = Math.floor(Math.random() * 10);
   129→    }
   130→
   131→    const { contractType, barrier, stake } = params;
   132→    let won = false;
   133→    switch (contractType) {
   134→      case 'DIGITMATCH': won = nextDigit === barrier; break;
   135→      case 'DIGITDIFF': won = nextDigit !== barrier; break;
   136→      case 'DIGITOVER': won = nextDigit > (barrier ?? 4); break;
   137→      case 'DIGITUNDER': won = nextDigit < (barrier ?? 5); break;
   138→      case 'DIGITEVEN': won = nextDigit % 2 === 0; break;
   139→      case 'DIGITODD': won = nextDigit % 2 === 1; break;
   140→      default: won = Math.random() > 0.5;
   141→    }
   142→
   143→    const isMatch = contractType === 'DIGITMATCH';
   144→    const profitMultiplier = isMatch ? 8.5 : 0.85;
   145→    const profit = won ? stake * profitMultiplier : -stake;
   146→    const payout = won ? stake + profit : 0;
   147→
   148→    logFn(`[SIM] ${contractType} ${params.symbol} d${barrier ?? '-'} $${stake} → ${won ? 'WIN +$' + profit.toFixed(2) : 'LOSS -$' + stake.toFixed(2)} (digit=${nextDigit})`);
   149→
   150→    return {
   151→      id: `SIM-${Date.now()}`,
   152→      type: contractType,
   153→      symbol: params.symbol,
   154→      stake,
   155→      payout,
   156→      profit,
   157→      digit: barrier ?? -1,
   158→      won,
   159→      timestamp: Date.now(),
   160→      simulated: true,
   161→    };
   162→  } catch (simErr) {
   163→    logFn(`[ERROR] SIM also failed: ${(simErr as Error).message}`);
   164→    return null;
   165→  }
   166→}

   501→    const runLoop = async () => {

   503→      await runCycle();
   504→      if (runningRef.current) {
   505→        // v8: 2s cycle — faster trading
   506→        cycleTimerRef.current = setTimeout(runLoop, 2000);
   507→      }
   508→    };
   509→    runLoop();
   510→  }, [isAuthorized, botConfig, addAutoTraderLog, runCycle, setGlobalAIRunning, setGlobalAICycleCount, setGlobalAITotalTrades, setGlobalAITotalProfit, setGlobalAILearningStats]);
   511→
   512→  const stopBot = useCallback(() => {
   513→    runningRef.current = false;
   514→    setGlobalAIRunning(false);
   515→    setGlobalAIStatus('idle');
   516→    if (cycleTimerRef.current) { clearTimeout(cycleTimerRef.current); cycleTimerRef.current = null; }
   517→    activeTradesRef.current.clear();
   518→    tradeLocksRef.current.clear();
   519→    pendingSimTradesGlobal.clear();
   520→    addAutoTraderLog(`[AI] ═══════════════════════════════════════`);
   521→    addAutoTraderLog(`[AI] === GLOBAL AI v5 STOPPED ===`);
   522→    addAutoTraderLog(`[AI] Cycles: ${cycleCountRef.current} | Trades: ${totalTradesRef.current} | P/L: ${totalProfitRef.current >= 0 ? '+' : ''}$${totalProfitRef.current.toFixed(2)}`);
   523→    addAutoTraderLog(`[AI] Session W/L: ${sessionWinsRef.current}/${sessionLossesRef.current}`);
   524→    addAutoTraderLog(`[AI] ═══════════════════════════════════════`);
   525→    aiEngine.saveLearningData();
   526→  }, [addAutoTraderLog, setGlobalAIRunning, setGlobalAIStatus]);
   527→
   528→  // Expose start/stop for AI Scanner buttons
   529→  useEffect(() => {
   530→    (window as any).__globalAI = { startBot, stopBot };
   531→    return () => { delete (window as any).__globalAI; };
   532→  }, [startBot, stopBot]);
   533→
   534→  // === Init: start scanning + auto-start bot ===
   535→  useEffect(() => {
   536→    mountedRef.current = true;
   537→    aiEngine.loadLearningData();
   538→    setGlobalAILearningStats(aiEngine.getLearningStats());
   539→    startMultiMarketScan();
   540→
   541→    // v11: Restore Deriv credentials and start bot regardless of mode
   542→    const store = useWorldpadStore.getState();
   543→    console.log('[GlobalAI] INIT — isAuthorized:', store.isAuthorized, 'mode:', store.accountMode, 'hasDemoToken:', !!store.demoToken, 'hasRealToken:', !!store.realToken, 'accountId:', store.selectedAccountId);
   544→
   545→    const initAndStart = async () => {
   546→      // v12: Hardcoded token from .env.local — always connect LIVE
   547→      const envToken = process.env.NEXT_PUBLIC_DERIV_TOKEN;
   548→      const envAppId = process.env.NEXT_PUBLIC_DERIV_APP_ID || '1089';
   549→      let connected = false;
   550→
   551→      // ALWAYS log token status for diagnostics
   552→      if (envToken) {
   553→        addAutoTraderLog(`[AUTH] Token found: ${envToken.slice(0, 12)}... (${envToken.length} chars)`);
   554→      } else {
   555→        addAutoTraderLog(`[AUTH] NO TOKEN FOUND in env — will run in SIM mode`);
   556→      }
   557→
   558→      if (envToken) {
   559→        addAutoTraderLog(`[AUTH] Connecting to Deriv WebSocket...`);
   560→        console.log('[GlobalAI] Using hardcoded token from .env.local');
   561→        const result = await restoreCredentials(envToken, envAppId);
   562→        if (result) {
   563→          console.log('[GlobalAI] ✅ LIVE connected:', result.loginid, 'balance:', result.balance, 'type:', result.accountType);
   564→          addAutoTraderLog(`[AUTH] ✅ CONNECTED — Account: ${result.loginid} | Balance: $${result.balance.toFixed(2)} ${result.currency} | Mode: ${result.accountType.toUpperCase()}`);
   565→          // Force store into authorized LIVE state
   566→          const s = useWorldpadStore.getState();
   567→          s.setBalance(result.balance);
   568→          s.setAccountInfo({ fullname: result.fullname, loginid: result.loginid, balance: result.balance, currency: result.currency });
   569→          s.setAccountMode(result.accountType);
   570→          s.setSelectedAccountId(result.loginid);
   571→          if (result.accountType === 'demo') s.setDemoToken(envToken); else s.setRealToken(envToken);
   572→          s.setDerivAppId(envAppId);
   573→          s.setIsAuthorized(true);
   574→          connected = true;
   575→        } else {
   576→          console.error('[GlobalAI] ❌ Hardcoded token FAILED to connect');
   577→          addAutoTraderLog(`[AUTH] ❌ TOKEN REJECTED by Deriv — running in SIM mode`);
   578→          addAutoTraderLog(`[AUTH] Check: 1) Token has Trade scope 2) Account is verified 3) Token not expired`);
   579→        }
   580→      } else if (store.isAuthorized && (store.demoToken || store.realToken)) {
   581→        // Fallback: use token from store (user logged in via modal)
   582→        const token = store.accountMode === 'demo' ? store.demoToken : store.realToken;
   583→        if (token) {
   584→          const result = await restoreCredentials(token, store.derivAppId || '1089');
   585→          if (result) {
   586→            useWorldpadStore.getState().setBalance(result.balance);
   587→            useWorldpadStore.getState().setAccountInfo({ fullname: result.fullname, loginid: result.loginid, balance: result.balance, currency: result.currency });
   588→            connected = true;
   589→          }
   590→        }
   591→      }
   592→
   593→      // ALWAYS auto-start the bot
   594→      if (mountedRef.current && !runningRef.current) {
   595→        const isLive = useWorldpadStore.getState().isAuthorized;
   596→        console.log('[GlobalAI] Starting bot... mode=', isLive ? 'LIVE' : 'SIM');
   597→        addAutoTraderLog(`[AI] ═══════════════════════════════════════`);
   598→        addAutoTraderLog(`[AI] === BOT STARTED === mode=${isLive ? 'LIVE' : 'SIM'}${connected ? ' ✅ CONNECTED' : ''}`);
   599→        addAutoTraderLog(`[AI] ═══════════════════════════════════════`);
   600→        startBot();
   601→      }
   602→    };
   603→
   604→    // Wait 5s for scanner to collect some data, then start
   605→    const autoStartTimer = setTimeout(initAndStart, 5000);
   606→
   607→    return () => {
   608→      mountedRef.current = false;
   609→      runningRef.current = false;
   610→      if (cycleTimerRef.current) clearTimeout(cycleTimerRef.current);
   611→      if (rankingTimerRef.current) clearTimeout(rankingTimerRef.current);
   612→      clearTimeout(autoStartTimer);
   613→      pendingSimTradesGlobal.clear();
   614→      stopMultiMarketScan();
   615→    };
   616→  }, []);
   617→
   618→  // Health monitoring
   619→  useEffect(() => {
   620→    const interval = setInterval(() => {
   621→      if (!mountedRef.current) return;
   622→      const connected = isScannerConnected();
   623→      const health = getScannerHealth();
   624→      if (mountedRef.current) {
   625→        setGlobalAIHealth({
   626→          isConnected: connected,
   627→          totalTicksReceived: health.totalTicksReceived,
   628→          lastTickTime: health.lastTickTime,
   629→          connectTime: health.connectTime,
   630→          ticksPerMarket: health.ticksPerMarket,
   631→          wsError: health.wsError,
   632→          callbackCount: health.callbackCount,
   633→        });
   634→      }
   635→    }, 1000);
   636→    return () => clearInterval(interval);
   637→  }, [setGlobalAIHealth]);
   638→
   639→  // v5: Auto-restart bot when user connects mid-session
   640→  // CRITICAL FIX: If bot is already running in SIM mode, we MUST stop + restart it
   641→  // so it picks up the new isAuthorized=true closure for LIVE trades.
   642→  useEffect(() => {
   643→    if (!isAuthorized) return;
   644→
   645→    const store = useWorldpadStore.getState();
   646→    const token = store.accountMode === 'demo' ? store.demoToken : store.realToken;
   647→    if (token) {
   648→      restoreCredentials(token, '1089');
   649→    }
   650→
   651→    // Start the bot (with credentials restored above)
   652→    if (runningRef.current) {
   653→      runningRef.current = false;
   654→      if (cycleTimerRef.current) { clearTimeout(cycleTimerRef.current); cycleTimerRef.current = null; }
   655→      activeTradesRef.current.clear();
   656→      tradeLocksRef.current.clear();
   657→      pendingSimTradesGlobal.clear();
   658→      setGlobalAIRunning(false);
   659→    }
   660→    addAutoTraderLog(`[AI] 🔗 Account connected — starting bot (${store.accountMode})...`);
   661→    setTimeout(() => { startBot(); }, 1000);
   662→  }, [isAuthorized, startBot, addAutoTraderLog, setGlobalAIRunning]);
   663→
   664→  // Listen for trades from ANY tab and feed AI learning
   665→  useEffect(() => {
   666→    const unsub = useWorldpadStore.subscribe(
   667→      (state) => state.tradeHistory,
   668→      (history) => {
   669→        if (history.length === 0) return;
   670→        const last = history[history.length - 1];
   671→        if (last.id.startsWith('ai-')) return;
   672→        aiEngine.recordTradeResult(
   673→          last.symbol, last.type, last.digit >= 0 ? last.digit : undefined,
   674→          last.profit, 50
   675→        );
   676→      }
   677→    );
   678→    return unsub;
   679→  }, []);
   680→
   681→  return null;
