    18→// === Pending simulation trades awaiting next tick (for GlobalAI direct trades) ===
    19→const pendingSimTradesGlobal: Map<string, {
    20→  contractType: string;
    21→  barrier: number | undefined;
    22→  stake: number;
    23→  symbol: string;
    24→  resolve: (result: TradeResult | null) => void;

    27→let globalTickListenerRegistered = false;
    28→
    29→function registerGlobalTickListener() {
    30→  if (globalTickListenerRegistered) return;
    31→  globalTickListenerRegistered = true;
    32→
    33→  addTickListener((symbol, data) => {
    34→    const pending = pendingSimTradesGlobal.get(symbol);
    35→    if (!pending || !data.lastTick) return;
    36→
    37→    const nextDigit = data.lastTick.digit;
    38→    pendingSimTradesGlobal.delete(symbol);
    39→
    40→    const { contractType, barrier, stake, symbol: tradeSymbol } = pending;
    41→    let won = false;
    42→    switch (contractType) {
    43→      case 'DIGITMATCH': won = nextDigit === barrier; break;
    44→      case 'DIGITDIFF': won = nextDigit !== barrier; break;
    45→      case 'DIGITOVER': won = nextDigit > (barrier ?? 4); break;
    46→      case 'DIGITUNDER': won = nextDigit < (barrier ?? 5); break;
    47→      case 'DIGITEVEN': won = nextDigit % 2 === 0; break;
    48→      case 'DIGITODD': won = nextDigit % 2 === 1; break;
    49→      default: won = Math.random() > 0.5;
    50→    }
    51→
    52→    // Payout: Deriv pays net profit multiplier on win, 0 on loss
    53→    const isMatch = contractType === 'DIGITMATCH';
    54→    const profitMultiplier = isMatch ? 8.5 : 0.85;
    55→    const profit = won ? stake * profitMultiplier : -stake;
    56→    const payout = won ? stake + profit : 0;
    57→
    58→    const result: TradeResult = {
    59→      id: `SIM-${Date.now()}`,
    60→      type: contractType,
    61→      symbol: tradeSymbol,
    62→      stake,
    63→      payout,
    64→      profit,
    65→      digit: barrier ?? -1,
    66→      won,
    67→      timestamp: Date.now(),
    68→      simulated: true,
    69→    };
    70→
    71→    pending.resolve(result);
    72→  });
    73→}

    75→async function placeTradeDirect(params: {
    76→  contractType: string;
    77→  barrier?: number;
    78→  stake: number;
    79→  symbol: string;
    80→}): Promise<TradeResult | null> {
    81→  // v5 FIX: Only check isAuthorized. isSimulating() refers to the single-market WS
    82→  // which is irrelevant — GlobalAI uses multi-market-ws for ticks (always real data).
    83→  const authorized = useWorldpadStore.getState().isAuthorized;
    84→  const simMode = !authorized;
    85→  const wsStatus = getTradeWSStatus();
    86→  console.log('[GlobalAI] placeTradeDirect: isAuthorized=', authorized, 'simMode=', simMode, 'ws=', wsStatus);
    87→  addAutoTraderLog(`[TRADE-MODE] authorized=${authorized} simMode=${simMode} wsReady=${wsStatus.wsReady} token=${wsStatus.tokenPreview}`);
    88→
    89→  if (simMode) {
    90→    console.warn('[GlobalAI] SIMULATION TRADE — NOT sending to Deriv. isAuthorized=', authorized);
    91→    addAutoTraderLog(`⚠️ SIM TRADE: ${params.contractType} ${params.symbol} $${params.stake} — NOT on Deriv`);
    92→    if (pendingSimTradesGlobal.has(params.symbol)) return null;
    93→
    94→    return new Promise<TradeResult | null>((resolve) => {
    95→      const timeout = setTimeout(() => {
    96→        pendingSimTradesGlobal.delete(params.symbol);
    97→        resolve(null);
    98→      }, 15000);
    99→
   100→      registerGlobalTickListener();
   101→      pendingSimTradesGlobal.set(params.symbol, {
   102→        ...params,
   103→        resolve: (r) => { clearTimeout(timeout); resolve(r); },
   104→      });
   105→    });
   106→  }
   107→
   108→  // LIVE mode — sending real trade to Deriv
   109→  try {
   110→    const wsBefore = getTradeWSStatus();
   111→    useWorldpadStore.getState().addAutoTraderLog(`[AI] 🔄 LIVE TRADE — connecting to Deriv... (ws ready: ${wsBefore.wsReady})`);
   112→    console.log('[GlobalAI] LIVE TRADE — sending to Deriv WS. Status:', wsBefore);
   113→
   114→    const proposal = await getProposalWS({
   115→      contractType: params.contractType,
   116→      symbol: params.symbol,
   117→      stake: params.stake,
   118→      barrier: params.barrier,
   119→      duration: 1,
   120→      durationUnit: 't',
   121→    });
   122→    useWorldpadStore.getState().addAutoTraderLog(`[AI] ✅ Proposal OK — $${proposal.payout.toFixed(2)} payout, buying...`);
   123→
   124→    const buyResult = await buyContractWS(proposal.id, proposal.ask_price);
   125→    const won = buyResult.profit > 0;
   126→    useWorldpadStore.getState().addAutoTraderLog(`[AI] ✅ BUY FILLED — contract ${buyResult.contract_id} | ${won ? '+$' : '-$'}${Math.abs(buyResult.profit).toFixed(2)} ${won ? 'WIN' : 'LOSS'} [LIVE ON DERIV]`);
   127→    return {
   128→      id: buyResult.contract_id,
   129→      type: params.contractType,
   130→      symbol: params.symbol,
   131→      stake: params.stake,
   132→      payout: buyResult.payout,
   133→      profit: buyResult.profit,
   134→      digit: params.barrier ?? -1,
   135→      won,
   136→      timestamp: Date.now(),
   137→      simulated: false,
   138→    };
   139→  } catch (err) {
   140→    const errMsg = (err as Error).message || String(err);
   141→    const wsAfter = getTradeWSStatus();
   142→    console.error('[GlobalAI] Live trade error:', errMsg, 'ws after:', wsAfter);
   143→    useWorldpadStore.getState().addAutoTraderLog(`[AI] ❌ TRADE FAILED: ${errMsg}`);
   144→    if (errMsg.includes('timed out')) {
   145→      useWorldpadStore.getState().addAutoTraderLog(`[AI] ❌ Trade timed out — check PAT token has TRADE scope in Deriv Settings`);
   146→    }
   147→    if (errMsg.includes('Not authenticated')) {
   148→      useWorldpadStore.getState().addAutoTraderLog(`[AI] ❌ Not connected — click Connect Account`);
   149→    }
   150→    return null;
   151→  }
   152→}

   162→// v2: Per-market loss cooldown tracker (shared with use-ai-bot)
   163→const lossCooldownsGlobal: Map<string, number> = new Map();
   164→const LOSS_COOLDOWN_TICKS = 4;
   165→const marketTickCountsGlobal: Map<string, number> = new Map();

   167→export function GlobalAI() {

   178→  const tradeLocksRef = useRef<Set<string>>(new Set());
   179→  const totalProfitRef = useRef(0);
   180→  const totalTradesRef = useRef(0);
   181→  const cycleCountRef = useRef(0);
   182→  const lastRankingRef = useRef<RankedMarket[]>([]);
   183→  const mountedRef = useRef(true);
   184→  const sessionWinsRef = useRef(0);
   185→  const sessionLossesRef = useRef(0);

   187→  // Init tick counters

   191→

   193→  const rankingTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   194→
   195→  const doUpdateRanking = useCallback(() => {

   205→    }

   208→  const updateRankingThrottled = useCallback(() => {
   209→    if (rankingTimerRef.current) return;
   210→    rankingTimerRef.current = setTimeout(() => {
   211→      rankingTimerRef.current = null;
   212→      doUpdateRanking();
   213→    }, 500);
   214→  }, [doUpdateRanking]);
   215→
   216→  // Tick callback — feeds AI + triggers ranking + manages cooldowns

   219→      feedTickToAI(symbol, data);
   220→      updateRankingThrottled();
   221→
   222→      // v2: Decrement cooldowns
   223→      const currentTicks = (marketTickCountsGlobal.get(symbol) || 0) + 1;
   224→      marketTickCountsGlobal.set(symbol, currentTicks);
   225→      const cdRemaining = lossCooldownsGlobal.get(symbol);
   226→      if (cdRemaining !== undefined && currentTicks >= cdRemaining) {
   227→        lossCooldownsGlobal.delete(symbol);
   228→      }
   229→    });
   230→    return unsubscribe;
   231→  }, [updateRankingThrottled]);
   232→
   233→  // v2: Build cooldown set
   234→  const getCooldownSet = useCallback((): Set<string> => {
   235→    const set = new Set<string>();
   236→    for (const [symbol, untilTick] of lossCooldownsGlobal) {
   237→      if ((marketTickCountsGlobal.get(symbol) || 0) < untilTick) set.add(symbol);
   238→    }
   239→    return set;
   240→  }, []);
   241→
   242→  // v2: Stop-loss check
   243→  const isStopLossHit = useCallback((): boolean => {
   244→    if (botConfig.stopLoss <= 0) return false;
   245→    return totalProfitRef.current <= -botConfig.stopLoss;

   247→
   248→  // === Trade execution ===
   249→  const executeTradeOnMarket = useCallback(async (market: RankedMarket, stake: number) => {
   250→    if (!market.selectedSignal) return;
   251→    if (tradeLocksRef.current.has(market.symbol)) return;
   252→
   253→    // v5: Real Money Guard — check safety rules
   254→    // v5 FIX: Read isAuthorized from store directly (not closure) to avoid stale values
   255→    const authorized = useWorldpadStore.getState().isAuthorized;
   256→    const simMode = !authorized;
   257→    console.log('[GlobalAI] executeTradeOnMarket: authorized=', authorized, 'simMode=', simMode, 'symbol=', market.symbol);
   258→    const guardResult = checkLiveTrade(stake, !simMode);
   259→    if (!guardResult.allowed) {

   261→      if (guardResult.warnings.length > 0) {
   262→        for (const w of guardResult.warnings) addAutoTraderLog(`[AI] WARNING: ${w}`);
   263→      }
   264→      return;
   265→    }
   266→    if (guardResult.warnings.length > 0) {
   267→      for (const w of guardResult.warnings) addAutoTraderLog(`[AI] WARNING: ${w}`);
   268→    }
   269→
   270→    tradeLocksRef.current.add(market.symbol);
   271→
   272→    try {
   273→      const signal = market.selectedSignal;
   274→      // v4: Dynamic stake via Kelly criterion
   275→      const { stake: kellyStake, reason: stakeReason } = calculateStake(
   276→        market.symbol, signal.contractType, 0.90, { baseStake: guardResult.cappedStake }
   277→      );
   278→      let finalStake = kellyStake > 0 ? kellyStake : guardResult.cappedStake;
   279→
   280→      // v5: Re-apply live stake caps after Kelly adjustment
   281→      if (!simMode) {
   282→        finalStake = Math.max(0.35, Math.min(5.0, finalStake));
   283→      }
   284→
   285→      const logMsg = `[AI] ${market.name}: ${signal.contractType} d${signal.barrier ?? '-'} @ $${finalStake.toFixed(2)}${simMode ? ' ⚠️SIM' : ' LIVE'} | ${signal.reason} | score ${market.combinedScore.toFixed(0)} | ${stakeReason}`;
   286→      addAutoTraderLog(logMsg);
   287→
   288→      activeTradesRef.current.set(market.symbol, { signal, startedAt: Date.now() });
   289→
   290→      const result = await placeTradeDirect({
   291→        contractType: signal.contractType,
   292→        barrier: signal.barrier,
   293→        stake: finalStake,
   294→        symbol: market.symbol,
   295→      });
   296→
   297→      if (result) {
   298→        const won = result.profit > 0;
   299→
   300→        // v5: Record live trade for safety tracking
   301→        if (!simMode) {
   302→          const liveResult = recordLiveTrade(result.profit);
   303→          if (liveResult.shouldPause) {
   304→            addAutoTraderLog(`[AI] PAUSED: ${liveResult.message}`);
   305→          }
   306→          if (liveResult.shouldStop) {
   307→            addAutoTraderLog(`[AI] HARD STOP: ${liveResult.message}`);
   308→            runningRef.current = false;
   309→            setGlobalAIRunning(false);
   310→            setGlobalAIStatus('idle');
   311→            aiEngine.saveLearningData();
   312→            return;
   313→          }
   314→        }
   315→
   316→        // v2: Per-market loss cooldown
   317→        if (!won) {
   318→          const ct = marketTickCountsGlobal.get(market.symbol) || 0;
   319→          lossCooldownsGlobal.set(market.symbol, ct + LOSS_COOLDOWN_TICKS);
   320→          sessionLossesRef.current++;
   321→          sessionWinsRef.current = 0;
   322→        } else {
   323→          sessionWinsRef.current++;
   324→          sessionLossesRef.current = 0;
   325→        }
   326→
   327→        addAutoTraderLog(won
   328→          ? `[AI] WIN  ${market.name}: +$${result.profit.toFixed(2)} | W:${sessionWinsRef.current} L:${sessionLossesRef.current}${!simMode ? ' [LIVE]' : ''}`
   329→          : `[AI] LOSS ${market.name}: $${result.profit.toFixed(2)} | W:${sessionWinsRef.current} L:${sessionLossesRef.current} | cooldown ${LOSS_COOLDOWN_TICKS} ticks${!simMode ? ' [LIVE]' : ''}`);
   330→
   331→        aiEngine.recordTradeResult(
   332→          market.symbol, signal.contractType, signal.barrier,
   333→          result.profit, market.combinedScore
   334→        );
   335→
   336→        // v4: Record to risk manager
   337→        recordRiskResult(market.symbol, result.profit);
   338→
   339→        addTradeResult({
   340→          id: `ai-${Date.now()}-${market.symbol}`,
   341→          type: signal.contractType,
   342→          symbol: market.symbol,
   343→          stake: finalStake,
   344→          payout: result.payout || finalStake * 0.85,
   345→          profit: result.profit,
   346→          digit: signal.barrier ?? -1,
   347→          won,
   348→          timestamp: Date.now(),
   349→        });
   350→
   351→        totalProfitRef.current += result.profit;
   352→        totalTradesRef.current += 1;
   353→        if (mountedRef.current) {
   354→          setGlobalAITotalTrades(totalTradesRef.current);
   355→          setGlobalAITotalProfit(totalProfitRef.current);
   356→          setGlobalAILearningStats(aiEngine.getLearningStats());
   357→        }
   358→      } else if (!simMode) {
   359→        // Trade returned null in LIVE mode — error already logged in placeTradeDirect
   360→        addAutoTraderLog(`[AI] ⚠️ Trade on ${market.name} returned no result (live path failed — see error above)`);
   361→      }
   362→    } catch (err) {
   363→      addAutoTraderLog(`[AI] Error on ${market.name}: ${(err as Error).message}`);
   364→    } finally {
   365→      activeTradesRef.current.delete(market.symbol);
   366→      tradeLocksRef.current.delete(market.symbol);
   367→    }
   368→  }, [isAuthorized, addAutoTraderLog, addTradeResult, setGlobalAITotalTrades, setGlobalAITotalProfit, setGlobalAILearningStats]);

   385→    setGlobalAIStatus('scanning');
   386→
   387→    const ranked = lastRankingRef.current;

   389→      setGlobalAIStatus('waiting');

   397→      const hasEv = m.expectedValue > 0;
   398→      const evStr = `score=${m.combinedScore.toFixed(0)} signal=${hasSignal} ev=${m.expectedValue.toFixed(3)}`;
   399→      if (!hasSignal) {
   400→        console.log(`[AI Cycle] SKIP ${m.name}: no signal (${evStr})`);
   401→      } else {
   402→        console.log(`[AI Cycle] OK ${m.name}: ${m.selectedSignal?.contractType} d${m.selectedSignal?.barrier} (${evStr})`);
   403→      }
   404→    }

   406→    // v2: Pass cooldown set
   407→    const cooldownSet = getCooldownSet();
   408→    const trades = selectTrades(ranked, {}, new Set(activeTradesRef.current.keys()), cooldownSet);

   410→      // v6: Log WHY no trades selected
   411→      const signalsExist = ranked.filter(m => m.selectedSignal).length;
   412→      const cooldownCount = cooldownSet.size;
   413→      const activeCount = activeTradesRef.current.size;
   414→      console.log(`[AI Cycle] NO TRADES — ranked=${ranked.length} withSignal=${signalsExist} cooldowns=${cooldownCount} active=${activeCount}`);
   415→      setGlobalAIStatus('waiting');
   416→      return;
   417→    }
   418→
   419→    setGlobalAIStatus('trading');
   420→    cycleCountRef.current += 1;
   421→    if (mountedRef.current) setGlobalAICycleCount(cycleCountRef.current);
   422→
   423→    // v2: Log EV summary for each trade
   424→    for (const t of trades) {
   425→      if (t.selectedSignal) {
   426→        addAutoTraderLog(`[AI] → ${t.name}: ${t.selectedSignal.contractType} d${t.selectedSignal.barrier ?? '-'} | EV=${t.expectedValue.toFixed(3)} | score=${t.combinedScore.toFixed(0)} | conf=${Math.round((t.selectedSignal.confidence || 0) * 100)}%`);
   427→      }
   428→    }
   429→
   430→    // Fire all trades in parallel (per-market locks prevent duplicates)
   431→    const promises = trades.map(trade => executeTradeOnMarket(trade, botConfig.stake));
   432→    await Promise.all(promises);
   433→
   434→    setGlobalAIStatus('waiting');
   435→  }, [executeTradeOnMarket, botConfig.stake, botConfig.stopLoss, setGlobalAIStatus, setGlobalAICycleCount, isStopLossHit, getCooldownSet]);

   437→  // === Start / Stop ===
   438→  const startBot = useCallback(() => {
   439→    // v2: Reset everything on fresh start
   440→    lossCooldownsGlobal.clear();
   441→    sessionWinsRef.current = 0;
   442→    sessionLossesRef.current = 0;
   443→    resetRiskStates(); // v4: reset Kelly staking
   444→
   445→    aiEngine.loadLearningData();
   446→    setGlobalAILearningStats(aiEngine.getLearningStats());
   447→    runningRef.current = true;
   448→    totalProfitRef.current = 0;
   449→    totalTradesRef.current = 0;
   450→    cycleCountRef.current = 0;
   451→    activeTradesRef.current.clear();
   452→    tradeLocksRef.current.clear();
   453→    setGlobalAIRunning(true);
   454→    setGlobalAICycleCount(0);
   455→    setGlobalAITotalTrades(0);
   456→    setGlobalAITotalProfit(0);
   457→
   458→    // v6 FIX: Read isAuthorized from store directly (not closure) to avoid stale values
   459→    const authorized = useWorldpadStore.getState().isAuthorized;
   460→    const simMode = !authorized;
   461→
   462→    // v5: Pre-flight safety check for live trading
   463→    if (!simMode) {
   464→      const preflight = preFlightCheck(true);
   465→      addAutoTraderLog(`[AI] ═══════════════════════════════════════`);
   466→      addAutoTraderLog(`[AI] === GLOBAL AI v5 STARTED === (LIVE MODE)`);

   482→      addAutoTraderLog(`[AI] ═══════════════════════════════════════`);
