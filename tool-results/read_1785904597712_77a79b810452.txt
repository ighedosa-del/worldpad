    97→    97→// ── Persistent Learning Storage ──
    98→    98→function loadLearning(): Map<string, ComboStats> {
    99→    99→  if (typeof window === 'undefined') return new Map();
   100→   100→  try {
   101→   101→    // Version check: clear old learning if strategy changed
   102→   102→    const version = parseInt(localStorage.getItem(LEARNING_VERSION_KEY) || '0');
   103→   103→    if (version < CURRENT_LEARNING_VERSION) {
   104→   104→      localStorage.removeItem(LEARNING_KEY);
   105→   105→      localStorage.setItem(LEARNING_VERSION_KEY, String(CURRENT_LEARNING_VERSION));
   106→   106→      return new Map(); // Fresh start with v7
   107→   107→    }
   108→   108→    const raw = localStorage.getItem(LEARNING_KEY);
   109→   109→    if (raw) {
   110→   110→      const arr: [string, ComboStats][] = JSON.parse(raw);
   111→   111→      return new Map(arr);
   112→   112→    }
   113→   113→  } catch {}
   114→   114→  return new Map();
   115→   115→}

   168→   168→function playTone(freq: number, duration: number, type: OscillatorType = 'sine', vol = 0.15) {
   169→   169→  try {
   170→   170→    const ctx = getAudioCtx();
   171→   171→    const osc = ctx.createOscillator();
   172→   172→    const gain = ctx.createGain();
   173→   173→    osc.type = type;
   174→   174→    osc.frequency.value = freq;
   175→   175→    gain.gain.value = vol;
   176→   176→    gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + duration);
   177→   177→    osc.connect(gain);
   178→   178→    gain.connect(ctx.destination);
   179→   179→    osc.start();
   180→   180→    osc.stop(ctx.currentTime + duration);
   181→   181→  } catch {}
   182→   182→}

   188→   188→// ── Contract mapping ──
   189→   189→// DIFFER IS BANNED — 0.09x payout means you lose $0.91 per loss vs gain $0.03 per win
   190→   190→function signalToContract(subType: string, prediction: number | null): { contractType: string; barrier: string } | null {
   191→   191→  const t = subType.toUpperCase();
   192→   192→  switch (t) {
   193→   193→    case 'MATCH': case 'DIGIT MATCH': return prediction !== null ? { contractType: 'DIGITMATCH', barrier: String(prediction) } : null;
   194→   194→    // DIFFER BANNED — returns null, trade will be skipped
   195→   195→    case 'DIFFER': case 'DIGIT DIFF': return null; // BANNED: 0.09x payout = guaranteed loss
   196→   196→    case 'OVER': case 'DIGIT OVER': return prediction !== null ? { contractType: 'DIGITOVER', barrier: String(prediction) } : null;
   197→   197→    case 'UNDER': case 'DIGIT UNDER': return prediction !== null ? { contractType: 'DIGITUNDER', barrier: String(prediction) } : null;
   198→   198→    case 'EVEN': return { contractType: 'DIGITEVEN', barrier: '' };
   199→   199→    case 'ODD': return { contractType: 'DIGITODD', barrier: '' };
   200→   200→    case 'RISE': return { contractType: 'CALL', barrier: '' };
   201→   201→    case 'FALL': return { contractType: 'PUT', barrier: '' };
   202→   202→    default: return null;
   203→   203→  }
   204→   204→}

   206→   206→// ── Stake Calculator (Simple + Anti-Martingale) ──
   207→   207→// Keep it simple: base stake most of the time
   208→   208→// Reduce after 3+ consecutive losses (survival)
   209→   209→// Increase after 3+ consecutive wins (ride the wave)
   210→   210→function calcStake(baseStake: number, consecutiveWins: number, consecutiveLosses: number, accountBalance: number, todayProfit: number, profitLockMode: boolean): number {
   211→   211→  let stake = baseStake;
   212→   212→  // Reduce after losses (but not too aggressive — was killing profits)
   213→   213→  if (consecutiveLosses >= 4) stake = baseStake * 0.5;
   214→   214→  else if (consecutiveLosses >= 3) stake = baseStake * 0.75;
   215→   215→  // Increase after wins
   216→   216→  else if (consecutiveWins >= 4) stake = baseStake * 1.4;
   217→   217→  else if (consecutiveWins >= 3) stake = baseStake * 1.25;
   218→   218→  else if (consecutiveWins >= 2) stake = baseStake * 1.15;
   219→   219→  else if (consecutiveWins >= 1) stake = baseStake * 1.05;
   220→   220→  // Profit lock
   221→   221→  if (profitLockMode && todayProfit >= 2) {
   222→   222→    stake = Math.min(stake, baseStake * 0.5);
   223→   223→  }
   224→   224→  // Safety: never risk more than 5% of balance, min $0.10
   225→   225→  const maxStake = Math.min(accountBalance * 0.05, 5);
   226→   226→  return Math.round(Math.max(stake, 0.10) * 100) / 100;
   227→   227→}

   411→   411→          } catch (innerErr) {

   413→   413→            return prev;
   414→   414→          }
   415→   415→        });
   416→   416→
   417→   417→        setStats(prev => ({
   418→   418→          ...prev,
   419→   419→          todayTrades: prev.todayTrades + 1,
   420→   420→          todayWins: prev.todayWins + (isWin ? 1 : 0),
   421→   421→          todayLosses: prev.todayLosses + (isWin ? 0 : 1),
   422→   422→          todayProfit: prev.todayProfit + safeProfit,
   423→   423→          consecutiveLosses: isWin ? 0 : prev.consecutiveLosses + 1,
   424→   424→          totalTrades: prev.totalTrades + 1,
   425→   425→          totalWins: prev.totalWins + (isWin ? 1 : 0),
   426→   426→          totalProfit: prev.totalProfit + safeProfit,
   427→   427→        }));
   428→   428→
   429→   429→        if (isWin) {
   430→   430→          consecutiveWinsRef.current++;
   431→   431→          setConsecutiveWins(consecutiveWinsRef.current);
   432→   432→        } else {
   433→   433→          consecutiveWinsRef.current = 0;
   434→   434→          setConsecutiveWins(0);
   435→   435→        }
   436→   436→
   437→   437→        // No auto-stop — trade non-stop forever
   438→   438→      }
   439→   439→      if (event === 'trade_error') setStatusMsg(`Trade error: ${d.message}`);
   440→   440→      } catch (err) {
   441→   441→        console.error('[EventHandler] Unhandled error:', err);
   442→   442→      }
   443→   443→    };
   444→   444→
   445→   445→    try {
   446→   446→      const client = new DerivClient(handler);
   447→   447→      clientRef.current = client;
   448→   448→      await client.connect(token, appId);
   449→   449→    } catch (err) {
   450→   450→      if (mountedRef.current) { setConnectError((err as Error).message); setConnecting(false); }
   451→   451→    }
   452→   452→  }, [updateComboStats, addPLPoint]);
   453→   453→
   454→   454→  const disconnect = useCallback(() => {
   455→   455→    clientRef.current?.disconnect();
   456→   456→    clientRef.current = null;
   457→   457→    setAccount(null);
   458→   458→    setSettings(prev => ({ ...prev, autoTrading: false }));
   459→   459→    setStatusMsg('Disconnected');
   460→   460→  }, []);
   461→   461→
   462→   462→  const symbolNameMap: Record<string, string> = {
   463→   463→    R_10: 'Vol 10', R_25: 'Vol 25', R_50: 'Vol 50',
   464→   464→    R_75: 'Vol 75', R_100: 'Vol 100',
   465→   465→    '1HZ10V': '1s V10', '1HZ25V': '1s V25',
   466→   466→    '1HZ50V': '1s V50', '1HZ75V': '1s V75',

   468→   468→  };
   469→   469→
   470→   470→  const processAllSignals = useCallback((signalMap: Map<string, DigitSignal>) => {
   471→   471→    const s = settingsRef.current;
   472→   472→    if (!s.autoTrading || !clientRef.current || !account) return;
   473→   473→
   474→   474→    const st = statsRef.current;
   475→   475→    // No daily trade limit — trade non-stop
   476→   476→
   477→   477→    // No daily loss stop — keep trading
   478→   478→
   479→   479→    // No smart pause — trade immediately after losses
   480→   480→
   481→   481→    // ═══ MAX CONCURRENT OPEN TRADES: 10 at a time ═══
   482→   482→    if (openTradeCountRef.current >= 10) return;
   483→   483→

   485→   485→    const stake = calcStake(s.baseStake, consecutiveWinsRef.current, st.consecutiveLosses, account.balance, st.todayProfit, s.profitLockMode);
   486→   486→
   487→   487→    let tradesThisCycle = 0;
   488→   488→    signalMap.forEach((signal, key) => {
   489→   489→      if (tradesThisCycle >= 10) return; // Up to 10 trades per cycle — scan all tabs at once
   490→   490→
   491→   491→      // Parse key
   492→   492→      let symbol: string;
   493→   493→      let tab: string;
   494→   494→      if (key.startsWith('R_')) {
   495→   495→        const idx = key.indexOf('_', 2);
   496→   496→        symbol = idx > 0 ? key.substring(0, idx) : key;
   497→   497→        tab = idx > 0 ? key.substring(idx + 1) : signal.tab;
   498→   498→      } else {
   499→   499→        const firstUs = key.indexOf('_');
   500→   500→        symbol = firstUs > 0 ? key.substring(0, firstUs) : key;
   501→   501→        tab = firstUs > 0 ? key.substring(firstUs + 1) : signal.tab;
   502→   502→      }
   503→   503→
   504→   504→      // Confidence thresholds per tab (v7 — MATCH-first, ultra-selective):
   505→   505→      // MATCH: 45%+ (low bar — 9x payout means even 12% actual WR profits)
   506→   506→      // Even/Odd: 65%+ (high bar — only 0.95x payout, need genuine edge)
   507→   507→      // Over/Under: 65%+ (high bar — only 0.85x payout)
   508→   508→      // Rise/Fall: 65%+ (high bar — only 0.85x payout)
   509→   509→      let effMinConfidence = 65;
   510→   510→      if (tab === 'matches_differs') effMinConfidence = 45;
   511→   511→      const effMinQuality = (tab === 'even_odd' || tab === 'over_under') ? 15 : 25;
   512→   512→
   513→   513→      if (signal.phase !== 'trade_now') return;
   514→   514→      if (signal.confidence < effMinConfidence) return;
   515→   515→      if (signal.qualityScore < effMinQuality) return;
   516→   516→
   517→   517→      const comboKey = `${symbol}_${tab}`;
   518→   518→
   519→   519→      // ═══ FAST LEARNING ENGINE ═══
   520→   520→      const combo = comboStatsRef.current.get(comboKey);
   521→   521→      const comboWins = combo?.wins ?? 0;
   522→   522→      const comboLosses = combo?.losses ?? 0;
   523→   523→      const comboTotal = comboWins + comboLosses;
   524→   524→      const comboWinRate = comboTotal > 0 ? Math.round((comboWins / comboTotal) * 100) : 0;
   525→   525→      const recentResults = combo?.recentResults ?? [];
   526→   526→
   527→   527→      // ── STREAK ANALYSIS (last 5 results) ──
   528→   528→      const last5 = recentResults.slice(-5);
   529→   529→      const recentWins = last5.filter(r => r === 1).length;
   530→   530→      const recentLosses = last5.filter(r => r === 0).length;
   531→   531→      const recentWinRate = last5.length > 0 ? Math.round((recentWins / last5.length) * 100) : 50;
   532→   532→
   533→   533→      // ── HARD BLOCK: skip combos that are actively losing ──
   534→   534→      // Block if: last 3+ results are ALL losses (hot streak of losses)
   535→   535→      if (last5.length >= 3) {
   536→   536→        const last3 = last5.slice(-3);
   537→   537→        if (last3.every(r => r === 0)) return; // 3 consecutive losses → skip entirely
   538→   538→      }
   539→   539→
   540→   540→      // ── TIMEOUT after loss: wait longer if combo is losing ──
   541→   541→      let cooldownTime = 2000;
   542→   542→      if (comboTotal >= 2) {
   543→   543→        if (comboWinRate < 35) cooldownTime = 30000;      // Terrible: 30s cooldown
   544→   544→        else if (comboWinRate < 45) cooldownTime = 15000;  // Bad: 15s cooldown
   545→   545→        else if (recentWinRate < 30 && last5.length >= 3) cooldownTime = 20000; // Recent slump
   546→   546→      }
   547→   547→
   548→   548→      // ── DYNAMIC CONFIDENCE THRESHOLD based on learning ──
   549→   549→      // MATCH: starts at 45%, adjusts with learning
   550→   550→      // Others: start at 65%, adjust with learning
   551→   551→      let dynamicThreshold = tab === 'matches_differs' ? 45 : 65;
   552→   552→
   553→   553→      if (comboTotal >= 2) {
   554→   554→        if (comboWinRate < 30) dynamicThreshold += 20;        // Terrible: need 90%+ to retry
   555→   555→        else if (comboWinRate < 40) dynamicThreshold += 12;   // Bad: need much higher
   556→   556→        else if (comboWinRate < 50) dynamicThreshold += 5;    // Below avg: slightly higher
   557→   557→        else if (comboWinRate >= 65) dynamicThreshold -= 5;   // Good: lower bar
   558→   558→        else if (comboWinRate >= 75) dynamicThreshold -= 10;  // Great: much lower bar
   559→   559→      }
   560→   560→
   561→   561→      // Recent streak adjustment
   562→   562→      if (last5.length >= 3) {
   563→   563→        if (recentWinRate <= 20) dynamicThreshold += 15;      // Recent disaster
   564→   564→        else if (recentWinRate <= 40) dynamicThreshold += 5;   // Recent slump
   565→   565→      }
   566→   566→
   567→   567→      // Apply the dynamic threshold
   568→   568→      if (signal.confidence < dynamicThreshold) return;
   569→   569→
   570→   570→      // ── Cooldown check (after dynamic threshold) ──
   571→   571→      const lastTradeTime = pendingTradesRef.current.get(comboKey) ?? 0;
   572→   572→      if (Date.now() - lastTradeTime < cooldownTime) return;
   573→   573→
   574→   574→      const mapped = signalToContract(signal.subType, signal.prediction);
   575→   575→      if (!mapped) return;
   576→   576→
   577→   577→      // ═══ EV FILTER (simplified — no Bayesian complexity) ═══
   578→   578→      const payout = PAYOUT_MULTIPLIERS[mapped.contractType] ?? 0.85;
   579→   579→      // Use learned win rate if available (5+ trades), otherwise signal confidence
   580→   580→      const effectiveWinProb = comboTotal >= 5
   581→   581→        ? comboWins / comboTotal
   582→   582→        : signal.confidence / 100;
   583→   583→      const ev = (effectiveWinProb * payout) - ((1 - effectiveWinProb) * 1);
   584→   584→
   585→   585→      // Block negative EV trades
   586→   586→      if (ev < MIN_EV_THRESHOLD) return;
   587→   587→
   588→   588→      // MATCH gets a lower EV threshold since payout is huge
   589→   589→      // Even at 12% WR: EV = 0.12 * 9 - 0.88 = +0.20 (very profitable)
   590→   590→      // So for MATCH, we use a much lower EV bar
   591→   591→      if (mapped.contractType === 'DIGITMATCH' && ev >= -0.5) {
   592→   592→        // Allow slightly negative EV for MATCH — the 9x payout makes variance work in our favor
   593→   593→        // A MATCH that hits at 11% WR breaks even. At 15% WR it's hugely profitable.
   594→   594→        // The learning engine will shut down combos that consistently lose.
   595→   595→      }
   596→   596→
   597→   597→      pendingTradesRef.current.set(comboKey, Date.now());
   598→   598→      tradesThisCycle++;
   599→   599→      const tradeId = `${symbol}_${tab}_${Date.now()}`;
   600→   600→
   601→   601→      // ── Classify learning strength (fast: starts after 2 trades) ──
   602→   602→      let learnedStrength: string;
   603→   603→      if (comboTotal === 0) learnedStrength = 'NEW';
   604→   604→      else if (comboTotal < 2) learnedStrength = 'LEARNING';
   605→   605→      else if (comboWinRate >= 70) learnedStrength = 'VERY STRONG';
   606→   606→      else if (comboWinRate >= 60) learnedStrength = 'STRONG';
   607→   607→      else if (comboWinRate >= 50) learnedStrength = 'MODERATE';
   608→   608→      else learnedStrength = 'WEAK';

   611→   611→      let learningBoost = 0;

   614→   614→        else if (comboWinRate >= 65) learningBoost = 5;
   615→   615→        else if (comboWinRate >= 55) learningBoost = 2;
   616→   616→        else if (comboWinRate < 35) learningBoost = -10;
   617→   617→        else if (comboWinRate < 45) learningBoost = -5;
   618→   618→      }
   619→   619→      const adjustedConfidence = Math.min(100, Math.max(0, signal.confidence + learningBoost));
   620→   620→
   621→   621→      const record: TradeRecord = {
   622→   622→        id: tradeId, contractId: null, symbol,
   623→   623→        symbolName: symbolNameMap[symbol] || symbol,
   624→   624→        tradeType: mapped.contractType, prediction: mapped.barrier || signal.subType,
   625→   625→        confidence: signal.confidence, qualityScore: signal.qualityScore,
   626→   626→        stake, status: 'PENDING', profit: null, payout: null, buyPrice: null,
   627→   627→        longcode: '', timestamp: Date.now(), tab,
   628→   628→        learnedWinRate: comboWinRate,
   629→   629→        learnedWins: comboWins,
   630→   630→        learnedLosses: comboLosses,
   631→   631→        learnedStrength,
   632→   632→        adjustedConfidence,
   633→   633→        expectedValue: ev,
   634→   634→        payoutMultiplier: payout,
   635→   635→      };
   636→   636→
   637→   637→      setTrades(prev => [record, ...prev].slice(0, 200));
   638→   638→      const learnTag = comboTotal > 0

   644→   644→
   645→   645→      clientRef.current!.placeTrade(symbol, mapped.contractType, mapped.barrier, stake, signal.ticksRecommendation, 't')
   646→   646→        .then(result => {

   648→   648→            setTrades(prev => prev.map(t => t.id === tradeId ? { ...t, status: 'ERROR', error: result.error } : t));
   649→   649→            pendingTradesRef.current.delete(comboKey);

   651→   651→        });
   652→   652→    });
   653→   653→  }, [account]);
