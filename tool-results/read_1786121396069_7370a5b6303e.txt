    27→let globalTickListenerRegistered = false;
    28→
    29→function registerGlobalTickListener() {
    30→  if (globalTickListenerRegistered) return;
    31→  globalTickListenerRegistered = true;
    32→
    33→  addTickListener((symbol, data) => {
    34→    const pending = pendingSimTradesGlobal.get(symbol);
    35→    if (!pending || !data.lastTick) return;
    36→
    37→    const nextDigit = data.lastTick.digit;
    38→    pendingSimTradesGlobal.delete(symbol);
    39→
    40→    const { contractType, barrier, stake, symbol: tradeSymbol } = pending;
    41→    let won = false;
    42→    switch (contractType) {
    43→      case 'DIGITMATCH': won = nextDigit === barrier; break;
    44→      case 'DIGITDIFF': won = nextDigit !== barrier; break;
    45→      case 'DIGITOVER': won = nextDigit > (barrier ?? 4); break;
    46→      case 'DIGITUNDER': won = nextDigit < (barrier ?? 5); break;
    47→      case 'DIGITEVEN': won = nextDigit % 2 === 0; break;
    48→      case 'DIGITODD': won = nextDigit % 2 === 1; break;
    49→      default: won = Math.random() > 0.5;
    50→    }
    51→
    52→    // Payout: Deriv pays net profit multiplier on win, 0 on loss
    53→    const isMatch = contractType === 'DIGITMATCH';
    54→    const profitMultiplier = isMatch ? 8.5 : 0.85;
    55→    const profit = won ? stake * profitMultiplier : -stake;
    56→    const payout = won ? stake + profit : 0;
    57→
    58→    const result: TradeResult = {
    59→      id: `SIM-${Date.now()}`,
    60→      type: contractType,
    61→      symbol: tradeSymbol,
    62→      stake,
    63→      payout,
    64→      profit,
    65→      digit: barrier ?? -1,
    66→      won,
    67→      timestamp: Date.now(),
    68→      simulated: true,
    69→    };
    70→
    71→    pending.resolve(result);
    72→  });
    73→}

    75→async function placeTradeDirect(params: {
    76→  contractType: string;
    77→  barrier?: number;
    78→  stake: number;
    79→  symbol: string;
    80→}): Promise<TradeResult | null> {
    81→  const authorized = useWorldpadStore.getState().isAuthorized;
    82→  const logFn = useWorldpadStore.getState().addAutoTraderLog;
    83→
    84→  // === LIVE MODE ===
    85→  if (authorized) {
    86→    try {
    87→      const wsBefore = getTradeWSStatus();
    88→      logFn(`[LIVE] ${params.contractType} ${params.symbol} $${params.stake} d${params.barrier ?? '-'} | ws=${wsBefore.wsReady} auth=${wsBefore.authorized}`);
    89→
    90→      const proposal = await getProposalWS({
    91→        contractType: params.contractType,
    92→        symbol: params.symbol,
    93→        stake: params.stake,
    94→        barrier: params.barrier,
    95→        duration: 1,
    96→        durationUnit: 't',
    97→      });
    98→      logFn(`[LIVE] Proposal OK: id=${proposal.id} ask=$${proposal.ask_price} payout=$${proposal.payout}`);
    99→
   100→      const buyResult = await buyContractWS(proposal.id, proposal.ask_price);
   101→      const won = buyResult.profit > 0;
   102→      logFn(`[LIVE] ${won ? 'WIN' : 'LOSS'} $${Math.abs(buyResult.profit).toFixed(2)} contract=${buyResult.contract_id}`);
   103→      return {
   104→        id: buyResult.contract_id,
   105→        type: params.contractType,
   106→        symbol: params.symbol,
   107→        stake: params.stake,
   108→        payout: buyResult.payout,
   109→        profit: buyResult.profit,
   110→        digit: params.barrier ?? -1,
   111→        won,
   112→        timestamp: Date.now(),
   113→        simulated: false,
   114→      };
   115→    } catch (err) {
   116→      const errMsg = (err as Error).message || String(err);
   117→      logFn(`[LIVE FAILED] ${errMsg} → SIM fallback`);
   118→      // Fall through to SIM below
   119→    }
   120→  }
   121→
   122→  // === SIM MODE (or LIVE fallback) ===
   123→  logFn(`[SIM] Entering SIM path for ${params.symbol}...`);
   124→  try {
   125→    const md = getMarketData(params.symbol);
   126→    let nextDigit: number;
   127→    if (md?.lastTick) {
   128→      nextDigit = md.lastTick.digit;
   129→    } else {
   130→      nextDigit = Math.floor(Math.random() * 10);
   131→    }
   132→
   133→    const { contractType, barrier, stake } = params;
   134→    let won = false;
   135→    switch (contractType) {
   136→      case 'DIGITMATCH': won = nextDigit === barrier; break;
   137→      case 'DIGITDIFF': won = nextDigit !== barrier; break;
   138→      case 'DIGITOVER': won = nextDigit > (barrier ?? 4); break;
   139→      case 'DIGITUNDER': won = nextDigit < (barrier ?? 5); break;
   140→      case 'DIGITEVEN': won = nextDigit % 2 === 0; break;
   141→      case 'DIGITODD': won = nextDigit % 2 === 1; break;
   142→      default: won = Math.random() > 0.5;
   143→    }
   144→
   145→    const isMatch = contractType === 'DIGITMATCH';
   146→    const profitMultiplier = isMatch ? 8.5 : 0.85;
   147→    const profit = won ? stake * profitMultiplier : -stake;
   148→    const payout = won ? stake + profit : 0;
   149→
   150→    logFn(`[SIM] ${contractType} ${params.symbol} d${barrier ?? '-'} $${stake} → ${won ? 'WIN +$' + profit.toFixed(2) : 'LOSS -$' + stake.toFixed(2)} (digit=${nextDigit})`);
   151→
   152→    return {
   153→      id: `SIM-${Date.now()}`,
   154→      type: contractType,
   155→      symbol: params.symbol,
   156→      stake,
   157→      payout,
   158→      profit,
   159→      digit: barrier ?? -1,
   160→      won,
   161→      timestamp: Date.now(),
   162→      simulated: true,
   163→    };
   164→  } catch (simErr) {
   165→    logFn(`[ERROR] SIM also failed: ${(simErr as Error).message}`);
   166→    return null;
   167→  }
   168→}

   485→
   486→    setGlobalAIStatus('waiting');
   487→  }, [executeTradeOnMarket, setGlobalAIStatus, setGlobalAICycleCount, isStopLossHit, getCooldownSet]);
   488→  runCycleRef.current = runCycle; // v14: keep ref in sync
   489→
   490→  // === Start / Stop ===
   491→  const startBot = useCallback(() => {
   492→    // v2: Reset everything on fresh start
   493→    lossCooldownsGlobal.clear();
   494→    sessionWinsRef.current = 0;
   495→    sessionLossesRef.current = 0;
   496→    resetRiskStates(); // v4: reset Kelly staking
   497→    resetLiveSession(); // v9: clear persisted guard stats so stale data doesn't block trades
   498→
   499→    aiEngine.loadLearningData();
   500→    setGlobalAILearningStats(aiEngine.getLearningStats());
   501→    runningRef.current = true;
   502→    totalProfitRef.current = 0;
   503→    totalTradesRef.current = 0;
   504→    cycleCountRef.current = 0;
   505→    activeTradesRef.current.clear();
   506→    tradeLocksRef.current.clear();
   507→    setGlobalAIRunning(true);
   508→    setGlobalAICycleCount(0);
   509→    setGlobalAITotalTrades(0);
   510→    setGlobalAITotalProfit(0);
   511→
   512→    // v6 FIX: Read isAuthorized from store directly (not closure) to avoid stale values
   513→    const authorized = useWorldpadStore.getState().isAuthorized;
   514→    const simMode = !authorized;
   515→
   516→    // v5: Pre-flight safety check for live trading
   517→    addAutoTraderLog(`[AI] ═══════════════════════════════════════`);
   518→    addAutoTraderLog(`[AI] === BOT v14 STARTED === mode=${simMode ? 'SIM' : 'LIVE'} authorized=${authorized}`);
   519→    addAutoTraderLog(`[AI] ═════════════════════════════════════`);
   520→
   521→    // v14: Use refs so the loop ALWAYS calls the latest function versions.
   522→    // This fixes the stale closure bug where the loop ran the first-render's
   523→    // runCycle forever, missing all subsequent code updates and store changes.
   524→    const runLoop = async () => {
   525→      if (!runningRef.current) return;
   526→      try {
   527→        await runCycleRef.current!();
   528→      } catch (err) {
   529→        console.error('[GlobalAI] runCycle error:', err);
   530→        addAutoTraderLog(`[AI] ❌ runCycle error: ${(err as Error).message}`);
   531→      }
   532→      if (runningRef.current) {
   533→        cycleTimerRef.current = setTimeout(runLoop, 2000);
   534→      }
   535→    };
   536→    runLoop();
   537→  }, [addAutoTraderLog, setGlobalAIRunning, setGlobalAICycleCount, setGlobalAITotalTrades, setGlobalAITotalProfit, setGlobalAILearningStats]);
   538→
   539→  const stopBot = useCallback(() => {
   540→    runningRef.current = false;
   541→    setGlobalAIRunning(false);
   542→    setGlobalAIStatus('idle');
   543→    if (cycleTimerRef.current) { clearTimeout(cycleTimerRef.current); cycleTimerRef.current = null; }
   544→    activeTradesRef.current.clear();
   545→    tradeLocksRef.current.clear();
   546→    pendingSimTradesGlobal.clear();
   547→    addAutoTraderLog(`[AI] ═══════════════════════════════════════`);
   548→    addAutoTraderLog(`[AI] === GLOBAL AI v5 STOPPED ===`);
   549→    addAutoTraderLog(`[AI] Cycles: ${cycleCountRef.current} | Trades: ${totalTradesRef.current} | P/L: ${totalProfitRef.current >= 0 ? '+' : ''}$${totalProfitRef.current.toFixed(2)}`);
   550→    addAutoTraderLog(`[AI] Session W/L: ${sessionWinsRef.current}/${sessionLossesRef.current}`);
   551→    addAutoTraderLog(`[AI] ═══════════════════════════════════════`);
   552→    aiEngine.saveLearningData();
   553→  }, [addAutoTraderLog, setGlobalAIRunning, setGlobalAIStatus]);
   554→
   555→  // Expose start/stop for AI Scanner buttons
   556→  useEffect(() => {
   557→    (window as any).__globalAI = { startBot, stopBot };
   558→    return () => { delete (window as any).__globalAI; };
   559→  }, [startBot, stopBot]);
   560→
   561→  // === Init: start scanning + auto-start bot ===
   562→  useEffect(() => {
   563→    mountedRef.current = true;
   564→    aiEngine.loadLearningData();
   565→    setGlobalAILearningStats(aiEngine.getLearningStats());
   566→    startMultiMarketScan();
   567→
   568→    // v11: Restore Deriv credentials and start bot regardless of mode
   569→    const store = useWorldpadStore.getState();
   570→    console.log('[GlobalAI] INIT — isAuthorized:', store.isAuthorized, 'mode:', store.accountMode, 'hasDemoToken:', !!store.demoToken, 'hasRealToken:', !!store.realToken, 'accountId:', store.selectedAccountId);
   571→
   572→    const initAndStart = async () => {
   573→      // v12: Hardcoded token from .env.local — always connect LIVE
   574→      const envToken = process.env.NEXT_PUBLIC_DERIV_TOKEN;
   575→      const envAppId = process.env.NEXT_PUBLIC_DERIV_APP_ID || '1089';
   576→      let connected = false;
   577→
   578→      // ALWAYS log token status for diagnostics
   579→      if (envToken) {
   580→        addAutoTraderLog(`[AUTH] Token found: ${envToken.slice(0, 12)}... (${envToken.length} chars)`);
   581→      } else {
   582→        addAutoTraderLog(`[AUTH] NO TOKEN FOUND in env — will run in SIM mode`);
   583→      }
   584→
   585→      if (envToken) {
   586→        addAutoTraderLog(`[AUTH] Connecting to Deriv WebSocket...`);
   587→        console.log('[GlobalAI] Using hardcoded token from .env.local');
   588→        const result = await restoreCredentials(envToken, envAppId);
   589→        if (result) {
   590→          console.log('[GlobalAI] ✅ LIVE connected:', result.loginid, 'balance:', result.balance, 'type:', result.accountType);
   591→          addAutoTraderLog(`[AUTH] ✅ CONNECTED — Account: ${result.loginid} | Balance: $${result.balance.toFixed(2)} ${result.currency} | Mode: ${result.accountType.toUpperCase()}`);
   592→          // Force store into authorized LIVE state
   593→          const s = useWorldpadStore.getState();
   594→          s.setBalance(result.balance);
   595→          s.setAccountInfo({ fullname: result.fullname, loginid: result.loginid, balance: result.balance, currency: result.currency });
   596→          s.setAccountMode(result.accountType);
   597→          s.setSelectedAccountId(result.loginid);
   598→          if (result.accountType === 'demo') s.setDemoToken(envToken); else s.setRealToken(envToken);
   599→          s.setDerivAppId(envAppId);
   600→          s.setIsAuthorized(true);
   601→          connected = true;
   602→        } else {
   603→          console.error('[GlobalAI] ❌ Hardcoded token FAILED to connect');
   604→          addAutoTraderLog(`[AUTH] ❌ TOKEN REJECTED by Deriv — running in SIM mode`);
   605→          addAutoTraderLog(`[AUTH] Check: 1) Token has Trade scope 2) Account is verified 3) Token not expired`);
   606→          // v13 FIX: Explicitly set isAuthorized=false so bot doesn't think it's LIVE
   607→          useWorldpadStore.getState().setIsAuthorized(false);
   608→        }
   609→      } else if (store.isAuthorized && (store.demoToken || store.realToken)) {
   610→        // Fallback: use token from store (user logged in via modal)
   611→        const token = store.accountMode === 'demo' ? store.demoToken : store.realToken;
   612→        if (token) {
   613→          const result = await restoreCredentials(token, store.derivAppId || '1089');
   614→          if (result) {
   615→            useWorldpadStore.getState().setBalance(result.balance);
   616→            useWorldpadStore.getState().setAccountInfo({ fullname: result.fullname, loginid: result.loginid, balance: result.balance, currency: result.currency });
   617→            connected = true;
   618→          }
   619→        }
   620→      }
   621→
   622→      // ALWAYS auto-start the bot
   623→      if (mountedRef.current && !runningRef.current) {
   624→        const isLive = useWorldpadStore.getState().isAuthorized;
   625→        console.log('[GlobalAI] Starting bot... mode=', isLive ? 'LIVE' : 'SIM');
   626→        addAutoTraderLog(`[AI] ═══════════════════════════════════════`);
   627→        addAutoTraderLog(`[AI] === BOT STARTED === mode=${isLive ? 'LIVE' : 'SIM'}${connected ? ' ✅ CONNECTED' : ''}`);
   628→        addAutoTraderLog(`[AI] ═══════════════════════════════════════`);
   629→        startBot();
   630→      }
   631→    };
   632→
   633→    // Wait 5s for scanner to collect some data, then start
   634→    const autoStartTimer = setTimeout(initAndStart, 5000);
   635→
   636→    return () => {
   637→      mountedRef.current = false;
   638→      runningRef.current = false;
   639→      if (cycleTimerRef.current) clearTimeout(cycleTimerRef.current);
   640→      if (rankingTimerRef.current) clearTimeout(rankingTimerRef.current);
   641→      clearTimeout(autoStartTimer);
   642→      pendingSimTradesGlobal.clear();
   643→      stopMultiMarketScan();
   644→    };
   645→  }, []);
   646→
   647→  // Health monitoring
   648→  useEffect(() => {
   649→    const interval = setInterval(() => {
   650→      if (!mountedRef.current) return;
   651→      const connected = isScannerConnected();
   652→      const health = getScannerHealth();
   653→      if (mountedRef.current) {
   654→        setGlobalAIHealth({
   655→          isConnected: connected,
   656→          totalTicksReceived: health.totalTicksReceived,
   657→          lastTickTime: health.lastTickTime,
   658→          connectTime: health.connectTime,
   659→          ticksPerMarket: health.ticksPerMarket,
   660→          wsError: health.wsError,
   661→          callbackCount: health.callbackCount,
   662→        });
   663→      }
   664→    }, 1000);
   665→    return () => clearInterval(interval);
   666→  }, [setGlobalAIHealth]);
   667→
   668→  // v13: REMOVED the isAuthorized restart effect.
   669→  // This was causing a CRITICAL race condition:
   670→  //   1) initAndStart (t=5s) connects WS + sets isAuthorized=true + starts bot
   671→  //   2) This effect fires because isAuthorized changed
   672→  //   3) It calls restoreCredentials AGAIN → kills the WS initAndStart just connected
   673→  //   4) It stops the running bot and restarts it → bot loop of death
   674→  // initAndStart now handles EVERYTHING: auth, store update, and bot start.
   675→  // No secondary effect needed.
   676→
   677→  // Listen for trades from ANY tab and feed AI learning
   678→  useEffect(() => {
   679→    const unsub = useWorldpadStore.subscribe(
   680→      (state) => state.tradeHistory,
   681→      (history) => {
   682→        if (history.length === 0) return;
   683→        const last = history[history.length - 1];
   684→        if (last.id.startsWith('ai-')) return;
   685→        aiEngine.recordTradeResult(
   686→          last.symbol, last.type, last.digit >= 0 ? last.digit : undefined,
   687→          last.profit, 50
   688→        );
   689→      }
   690→    );
   691→    return unsub;
   692→  }, []);
   693→
   694→  return null;
   695→}
