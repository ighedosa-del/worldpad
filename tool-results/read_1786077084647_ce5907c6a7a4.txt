    27→    27→let globalTickListenerRegistered = false;
    28→    28→
    29→    29→function registerGlobalTickListener() {
    30→    30→  if (globalTickListenerRegistered) return;
    31→    31→  globalTickListenerRegistered = true;
    32→    32→
    33→    33→  addTickListener((symbol, data) => {
    34→    34→    const pending = pendingSimTradesGlobal.get(symbol);
    35→    35→    if (!pending || !data.lastTick) return;
    36→    36→
    37→    37→    const nextDigit = data.lastTick.digit;
    38→    38→    pendingSimTradesGlobal.delete(symbol);
    39→    39→
    40→    40→    const { contractType, barrier, stake, symbol: tradeSymbol } = pending;
    41→    41→    let won = false;
    42→    42→    switch (contractType) {
    43→    43→      case 'DIGITMATCH': won = nextDigit === barrier; break;
    44→    44→      case 'DIGITDIFF': won = nextDigit !== barrier; break;
    45→    45→      case 'DIGITOVER': won = nextDigit > (barrier ?? 4); break;
    46→    46→      case 'DIGITUNDER': won = nextDigit < (barrier ?? 5); break;
    47→    47→      case 'DIGITEVEN': won = nextDigit % 2 === 0; break;
    48→    48→      case 'DIGITODD': won = nextDigit % 2 === 1; break;
    49→    49→      default: won = Math.random() > 0.5;
    50→    50→    }
    51→    51→
    52→    52→    // Payout: Deriv pays net profit multiplier on win, 0 on loss
    53→    53→    const isMatch = contractType === 'DIGITMATCH';
    54→    54→    const profitMultiplier = isMatch ? 8.5 : 0.85;
    55→    55→    const profit = won ? stake * profitMultiplier : -stake;
    56→    56→    const payout = won ? stake + profit : 0;
    57→    57→
    58→    58→    const result: TradeResult = {
    59→    59→      id: `SIM-${Date.now()}`,
    60→    60→      type: contractType,
    61→    61→      symbol: tradeSymbol,
    62→    62→      stake,
    63→    63→      payout,
    64→    64→      profit,
    65→    65→      digit: barrier ?? -1,
    66→    66→      won,
    67→    67→      timestamp: Date.now(),
    68→    68→      simulated: true,
    69→    69→    };
    70→    70→
    71→    71→    pending.resolve(result);
    72→    72→  });
    73→    73→}

    87→    87→  addAutoTraderLog(`[TRADE-MODE] authorized=${authorized} simMode=${simMode} wsReady=${wsStatus.wsReady} token=${wsStatus.tokenPreview}`);
    88→    88→
    89→    89→  if (simMode) {
    90→    90→    console.warn('[GlobalAI] SIMULATION TRADE — NOT sending to Deriv. isAuthorized=', authorized);
    91→    91→    addAutoTraderLog(`⚠️ SIM TRADE: ${params.contractType} ${params.symbol} $${params.stake} — NOT on Deriv`);
    92→    92→    if (pendingSimTradesGlobal.has(params.symbol)) return null;
    93→    93→
    94→    94→    return new Promise<TradeResult | null>((resolve) => {
    95→    95→      const timeout = setTimeout(() => {
    96→    96→        pendingSimTradesGlobal.delete(params.symbol);
    97→    97→        resolve(null);
    98→    98→      }, 15000);
    99→    99→
   100→   100→      registerGlobalTickListener();
   101→   101→      pendingSimTradesGlobal.set(params.symbol, {
   102→   102→        ...params,
   103→   103→        resolve: (r) => { clearTimeout(timeout); resolve(r); },
   104→   104→      });
   105→   105→    });
   106→   106→  }
   107→   107→
   108→   108→  // LIVE mode — sending real trade to Deriv
   109→   109→  try {
   110→   110→    const wsBefore = getTradeWSStatus();
   111→   111→    useWorldpadStore.getState().addAutoTraderLog(`[AI] 🔄 LIVE TRADE — connecting to Deriv... (ws ready: ${wsBefore.wsReady})`);
   112→   112→    console.log('[GlobalAI] LIVE TRADE — sending to Deriv WS. Status:', wsBefore);
   113→   113→
   114→   114→    const proposal = await getProposalWS({
   115→   115→      contractType: params.contractType,
   116→   116→      symbol: params.symbol,
   117→   117→      stake: params.stake,
   118→   118→      barrier: params.barrier,
   119→   119→      duration: 1,
   120→   120→      durationUnit: 't',
   121→   121→    });
   122→   122→    useWorldpadStore.getState().addAutoTraderLog(`[AI] ✅ Proposal OK — $${proposal.payout.toFixed(2)} payout, buying...`);
   123→   123→
   124→   124→    const buyResult = await buyContractWS(proposal.id, proposal.ask_price);
   125→   125→    const won = buyResult.profit > 0;
   126→   126→    useWorldpadStore.getState().addAutoTraderLog(`[AI] ✅ BUY FILLED — contract ${buyResult.contract_id} | ${won ? '+$' : '-$'}${Math.abs(buyResult.profit).toFixed(2)} ${won ? 'WIN' : 'LOSS'} [LIVE ON DERIV]`);
   127→   127→    return {
   128→   128→      id: buyResult.contract_id,
   129→   129→      type: params.contractType,
   130→   130→      symbol: params.symbol,
   131→   131→      stake: params.stake,
   132→   132→      payout: buyResult.payout,
   133→   133→      profit: buyResult.profit,
   134→   134→      digit: params.barrier ?? -1,
   135→   135→      won,
   136→   136→      timestamp: Date.now(),
   137→   137→      simulated: false,
   138→   138→    };
   139→   139→  } catch (err) {
   140→   140→    const errMsg = (err as Error).message || String(err);
   141→   141→    const wsAfter = getTradeWSStatus();
   142→   142→    console.error('[GlobalAI] Live trade error:', errMsg, 'ws after:', wsAfter);
   143→   143→    useWorldpadStore.getState().addAutoTraderLog(`[AI] ❌ TRADE FAILED: ${errMsg}`);
   144→   144→    if (errMsg.includes('timed out')) {
   145→   145→      useWorldpadStore.getState().addAutoTraderLog(`[AI] ❌ Trade timed out — check PAT token has TRADE scope in Deriv Settings`);
   146→   146→    }
   147→   147→    if (errMsg.includes('Not authenticated')) {
   148→   148→      useWorldpadStore.getState().addAutoTraderLog(`[AI] ❌ Not connected — click Connect Account`);
   149→   149→    }
   150→   150→    return null;
   151→   151→  }
   152→   152→}

   410→   410→    // that has distribution data (avoids waiting forever for signals to build)

   417→   417→      if (forceTarget) {
