    27→// === Digit button ===
    28→function DigitButton({ digit, selected, onClick, dist, total }: {
    29→  digit: number; selected: boolean; onClick: () => void; dist: number; total: number;
    30→}) {
    31→  const pct = total > 0 ? (dist / total) * 100 : 10;
    32→  const hot = pct >= 13;
    33→  const cold = pct <= 7;
    34→  return (
    35→    <button
    36→      onClick={onClick}
    37→      className="relative flex flex-col items-center justify-center w-full aspect-square rounded-xl transition-all duration-200 group"
    38→      style={{
    39→        background: selected ? 'rgba(168,85,247,0.15)' : 'rgba(255,255,255,0.03)',
    40→        border: selected ? '1.5px solid rgba(168,85,247,0.6)' : '1px solid rgba(255,255,255,0.06)',
    41→        boxShadow: selected ? '0 0 16px rgba(168,85,247,0.2)' : hot ? `0 0 8px ${hot ? '#22c55e' : '#ef4444'}20` : 'none',
    42→      }}
    43→    >
    44→      <span
    45→        className="text-lg sm:text-xl font-black font-mono transition-all duration-200 group-hover:scale-110"
    46→        style={{
    47→          color: selected ? '#a855f7' : hot ? '#22c55e' : cold ? '#ef4444' : '#e2e8f0',
    48→          textShadow: selected ? '0 0 10px rgba(168,85,247,0.5)' : 'none',
    49→        }}
    50→      >
    51→        {digit}
    52→      </span>
    53→      <span className="text-[9px] font-mono mt-0.5" style={{ color: hot ? '#22c55e' : cold ? '#ef4444' : '#6b7280' }}>
    54→        {pct.toFixed(1)}%
    55→      </span>
    56→    </button>
    57→  );
    58→}

    60→// === Trade history row ===
    61→function TradeRow({ trade }: { trade: { id: string; type: string; symbol: string; stake: number; profit: number; won: boolean; timestamp: number; digit: number } }) {
    62→  const formatCT = (t: string) => {
    63→    const map: Record<string, string> = {
    64→      DIGITOVER: 'Over', DIGITUNDER: 'Under', DIGITMATCH: 'Match',
    65→      DIGITDIFF: 'Differ', DIGITEVEN: 'Even', DIGITODD: 'Odd',
    66→    };
    67→    return map[t] || t;
    68→  };
    69→  const fmtSymbol = (s: string) => s.replace('R_', 'V').replace('1HZ', 'V');
    70→  return (
    71→    <div className="flex items-center gap-2 px-3 py-2 rounded-lg" style={{ background: 'rgba(255,255,255,0.02)', border: '1px solid rgba(255,255,255,0.04)' }}>
    72→      {trade.won
    73→        ? <CheckCircle2 className="w-3.5 h-3.5 text-green-400 shrink-0" />
    74→        : <XCircle className="w-3.5 h-3.5 text-red-400 shrink-0" />}
    75→      <div className="flex-1 min-w-0 flex items-center gap-1.5">
    76→        <span className="text-[11px] font-bold text-white">{fmtSymbol(trade.symbol)}</span>
    77→        <span className="text-[10px] font-mono px-1.5 py-0.5 rounded" style={{
    78→          background: trade.won ? 'rgba(34,197,94,0.1)' : 'rgba(239,68,68,0.1)',
    79→          color: trade.won ? '#22c55e' : '#ef4444',
    80→        }}>{formatCT(trade.type)}{trade.digit >= 0 ? ` d${trade.digit}` : ''}</span>
    81→      </div>
    82→      <div className="text-right shrink-0">
    83→        <div className="text-xs font-bold font-mono" style={{ color: trade.profit >= 0 ? '#22c55e' : '#ef4444' }}>
    84→          {trade.profit >= 0 ? '+' : ''}${trade.profit.toFixed(2)}
    85→        </div>
    86→        <div className="text-[9px] text-gray-600">${trade.stake.toFixed(2)}</div>
    87→      </div>
    88→      <span className="text-[9px] text-gray-600 font-mono shrink-0">
    89→        {new Date(trade.timestamp).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit', second: '2-digit' })}
    90→      </span>
    91→    </div>
    92→  );
    93→}
